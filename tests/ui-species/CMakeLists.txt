cmake_minimum_required(VERSION 3.19)
project(ui_species CXX)
set(CMAKE_CXX_STANDARD 11)
get_filename_component(ROOT "../.." ABSOLUTE)
enable_testing()
function(extract source signature output)
  string(FIND "${source}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing ${signature}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} part)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/${output}.h" "${part}\n")
endfunction()
foreach(path ui_main.c ui_shared.c ui_local.h ui_shared.h)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/codemp/ui/${path}")
endforeach()
file(READ "${ROOT}/codemp/ui/ui_main.c" main)
file(READ "${ROOT}/codemp/ui/ui_shared.c" shared)
file(READ "${ROOT}/codemp/ui/ui_local.h" types)
string(FIND "${types}" "#define SKIN_LENGTH" start)
string(FIND "${types}" "} playerSpeciesInfo_t;" end)
string(LENGTH "} playerSpeciesInfo_t;" suffix)
math(EXPR length "${end}-${start}+${suffix}")
string(SUBSTRING "${types}" ${start} ${length} species_types)
file(READ "${ROOT}/codemp/ui/ui_shared.h" types)
string(REGEX MATCH "typedef struct multiDef_s[^}]+} multiDef_t;" multi_type "${types}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/types.h" "#define MAX_MULTI_CVARS 256\n${species_types}\n${multi_type}\n")
foreach(output lifecycle character shared count init script)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}.h" "")
endforeach()
foreach(signature "static void UI_FreeSpecies(" "void UI_FreeAllSpecies( void )\n{" "qboolean UI_EnsurePlayerSpecies(" "static const playerSpeciesInfo_t *UI_CurrentPlayerSpecies(" "void UI_EnsureSpeciesOptions(" "void UI_BuildPlayerModel_List(" "static void UI_InitSpeciesCatalog(")
  extract("${main}" "${signature}" lifecycle)
endforeach()
extract("${main}" "static void UI_ReadCharacterCvars(" character)
extract("${main}" "static qboolean UI_MenuUsesPlayerSpecies(" character)
extract("${main}" "void UI_GetCharacterCvars(" character)
extract("${main}" "static void UI_InitCharacterSelection(" character)
foreach(signature "qboolean Rect_ContainsPoint(" "int Item_Multi_CountSettings(" "int Item_Multi_FindCvarByValue(" "const char *Item_Multi_Setting(" "qboolean Item_Multi_HandleKey(" "qboolean ItemParse_cvarStrList(")
  extract("${shared}" "${signature}" shared)
endforeach()
foreach(signature "menuDef_t *Menu_GetScriptMenu(" "static void Item_RunMenuScript(" "void Item_RunScript(itemDef_t *item, const char *s)\n{" "void  Menus_Activate(")
  extract("${shared}" "${signature}" script)
endforeach()
# Compile the actual species feeder-count cases without unrelated server,
# siege, cosmetic and scoreboard service dependencies.
string(FIND "${main}" "static int UI_FeederCount(" start)
string(SUBSTRING "${main}" ${start} -1 count_tail)
string(FIND "${count_tail}" "case FEEDER_PLAYER_SPECIES:" start)
string(FIND "${count_tail}" "case FEEDER_SIEGE_BASE_CLASS:" end)
math(EXPR length "${end}-${start}")
string(SUBSTRING "${count_tail}" ${start} ${length} count_cases)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/count.h" "static int speciesCount(float feederID) { switch(int(feederID)) { ${count_cases} } return -1; }\n")
# Bind the startup fixture to the actual no-scan init path. Closed menu parsing
# and initial preview-cvar setup are exercised by the production functions.
string(FIND "${main}" "void UI_Init( qboolean inGameLoad )" start)
string(SUBSTRING "${main}" ${start} -1 init_tail)
string(FIND "${init_tail}" "#define\tUI_FPS_FRAMES" end)
string(SUBSTRING "${init_tail}" 0 ${end} init)
foreach(forbidden "UI_EnsurePlayerSpecies(" "UI_GetCharacterCvars(" "UI_BuildPlayerModel_List(" "UI_BuildPlayerSpeciesCatalog(")
  string(FIND "${init}" "${forbidden}" location)
  if(NOT location LESS 0)
    message(FATAL_ERROR "UI_Init eagerly reaches the species catalog: ${forbidden}")
  endif()
endforeach()
foreach(required "UI_InitSpeciesCatalog(lazySpecies);" "UI_InitCharacterSelection(lazySpecies);" "\"ui_lazyPlayerSpecies\", \"1\"")
  string(FIND "${init}" "${required}" location)
  if(location LESS 0)
    message(FATAL_ERROR "UI_Init missing lazy lifecycle step: ${required}")
  endif()
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/init.h" "UI_InitSpeciesCatalog(lazySpecies);\nUI_InitCharacterSelection(lazySpecies);\n")
add_executable(ui_species check.cpp)
target_include_directories(ui_species PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${ROOT}/codemp/ui")
if(MSVC)
  target_compile_options(ui_species PRIVATE /EHsc)
  target_compile_definitions(ui_species PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_test(NAME ui_species COMMAND ui_species)
