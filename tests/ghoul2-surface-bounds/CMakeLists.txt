cmake_minimum_required(VERSION 3.16)
project(ghoul2_surface_bounds CXX)
set(CMAKE_CXX_STANDARD 11)
get_filename_component(ROOT "../.." ABSOLUTE)
set(GHOUL2_TEST_SOURCE_DIR "${ROOT}/codemp" CACHE PATH "Renderer sources; can select a baseline fixture tree")
foreach(renderer dedicated vanilla rend2 vulkan)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${renderer}.h" "")
  foreach(pair "G2_surfaces.cpp|qboolean G2_RemoveSurface(" "G2_bolts.cpp|int G2_Add_Bolt_Surf_Num(")
    string(REPLACE "|" ";" pieces "${pair}")
    list(GET pieces 0 file)
    list(GET pieces 1 signature)
    set(path "${GHOUL2_TEST_SOURCE_DIR}/rd-${renderer}/${file}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
    file(READ "${path}" source)
    string(FIND "${source}" "${signature}" start)
    if(start LESS 0)
      message(FATAL_ERROR "Missing ${signature} in ${path}")
    endif()
    string(SUBSTRING "${source}" ${start} -1 tail)
    string(FIND "${tail}" "\n}" end)
    math(EXPR length "${end}+2")
    string(SUBSTRING "${tail}" 0 ${length} function)
    file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/${renderer}.h" "${function}\n")
  endforeach()
endforeach()
add_executable(ghoul2_surface_bounds check.cpp)
target_include_directories(ghoul2_surface_bounds PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
  target_compile_options(ghoul2_surface_bounds PRIVATE /EHsc)
endif()
enable_testing()
add_test(NAME ghoul2_surface_bounds COMMAND ghoul2_surface_bounds)
