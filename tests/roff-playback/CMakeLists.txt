get_filename_component(repo "../.." ABSOLUTE)
set(REFERENCE_REF "" CACHE STRING "Optional git ref to reproduce original ROFF failures")
function(source path output)
  if(REFERENCE_REF)
    execute_process(COMMAND git show "${REFERENCE_REF}:${path}" WORKING_DIRECTORY "${repo}"
      OUTPUT_VARIABLE text COMMAND_ERROR_IS_FATAL ANY)
  else()
    file(READ "${repo}/${path}" text)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${repo}/${path}")
  endif()
  set(${output} "${text}" PARENT_SCOPE)
endfunction()
source(codemp/qcommon/RoffSystem.h declarations)
# Keep the actual class and member types, exposing private fixture entry points.
# Its production server include is unnecessary for these isolated methods.
string(REPLACE "#include \"server/server.h\"" "" declarations "${declarations}")
string(REPLACE "\nprivate:\n" "\npublic:\n" declarations "${declarations}")
# Count the actual playback-state allocations to catch restart leaks on systems
# whose AddressSanitizer runtime does not provide leak detection.
set(entity_marker "struct SROFFEntity\n\t//--------------------------------------\n\t{")
set(entity_hooks "${entity_marker}\n    static void *operator new(size_t size) { void *p = ::operator new(size); ++liveROFFEntities; return p; }\n    static void operator delete(void *p) { --liveROFFEntities; ::operator delete(p); }")
string(FIND "${declarations}" "${entity_marker}" entity_start)
if(entity_start LESS 0)
  message(FATAL_ERROR "Missing SROFFEntity allocation hook location")
endif()
string(REPLACE "${entity_marker}" "${entity_hooks}" declarations "${declarations}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/roff_declarations.h" "${declarations}")
source(codemp/qcommon/RoffSystem.cpp implementation)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/roff_methods.h" "")
function(extract signature)
  string(FIND "${implementation}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing ${signature}")
  endif()
  string(SUBSTRING "${implementation}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} method)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/roff_methods.h" "${method}\n")
endfunction()
foreach(signature "CROFFSystem::CROFF::CROFF( const char *file, int id )"
    "\nCROFFSystem::CROFF::~CROFF()\n{" "qboolean CROFFSystem::Restart()"
    "qboolean CROFFSystem::InitROFF2(" "void CROFFSystem::FixBadAngles("
    "qboolean CROFFSystem::Play(" "qboolean CROFFSystem::PurgeEnt( int entID,"
    "void CROFFSystem::UpdateEntities(" "qboolean CROFFSystem::ApplyROFF("
    "void CROFFSystem::ProcessNote(" "qboolean CROFFSystem::ClearLerp("
    "void CROFFSystem::SetLerp(")
  extract("${signature}")
endforeach()
