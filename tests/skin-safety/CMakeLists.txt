cmake_minimum_required(VERSION 3.19)
project(skin_safety CXX)
set(CMAKE_CXX_STANDARD 11)
set(REFERENCE_REF "" CACHE STRING "Optional git reference to reproduce original failures")
get_filename_component(ROOT "../.." ABSOLUTE)
enable_testing()

function(extract source signature output)
  string(FIND "${source}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing production function: ${signature}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} function)
  file(APPEND "${output}" "${function}\n")
endfunction()
foreach(renderer vanilla vulkan rend2 dedicated rend2_jk2)
  if(renderer MATCHES "^rend2")
    set(path shared/rd-rend2/tr_skin.cpp)
  else()
    set(path codemp/rd-${renderer}/tr_skin.cpp)
  endif()
  if(REFERENCE_REF)
    execute_process(COMMAND git show "${REFERENCE_REF}:${path}" WORKING_DIRECTORY "${ROOT}"
      OUTPUT_VARIABLE source COMMAND_ERROR_IS_FATAL ANY)
  else()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/${path}")
    file(READ "${ROOT}/${path}" source)
  endif()
  set(output "${CMAKE_CURRENT_BINARY_DIR}/${renderer}.h")
  file(WRITE "${output}" "")
  foreach(signature "static char *CommaParse( char **data_p ) {" "bool RE_SplitSkins(" "qhandle_t RE_RegisterIndividualSkin(" "qhandle_t RE_RegisterSkin(")
    extract("${source}" "${signature}" "${output}")
  endforeach()
  add_executable(skin_${renderer} check.cpp)

  target_compile_definitions(skin_${renderer} PRIVATE SKIN_SOURCE="${renderer}.h")
  if(renderer STREQUAL rend2_jk2)
    target_compile_definitions(skin_${renderer} PRIVATE JK2_MODE)
  endif()

    target_compile_definitions(skin_${renderer} PRIVATE TEST_SPLIT_NO_SIZE)
  endif()

    target_compile_definitions(skin_${renderer} PRIVATE _CRT_SECURE_NO_WARNINGS)
  endif()
  add_test(NAME skin_${renderer} COMMAND skin_${renderer})
endforeach()
