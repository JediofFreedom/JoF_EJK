cmake_minimum_required(VERSION 3.19)
project(shader_text_cache CXX)
set(CMAKE_CXX_STANDARD 11)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")

function(extract source signature out)
  string(FIND "${source}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing ${signature}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} result)
  set(${out} "${result}" PARENT_SCOPE)
endfunction()

foreach(path codemp/rd-vulkan/tr_shader.cpp codemp/qcommon/q_shared.c shared/qcommon/q_string.c)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/${path}")
endforeach()
file(READ "${ROOT}/codemp/qcommon/q_shared.c" parser)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/parser.h" "")
foreach(signature "const char *SkipWhitespace(" "char *COM_ParseExt(" "qboolean SkipBracedSection (" "void SkipRestOfLine (" "void COM_BeginParseSession(" "int COM_GetCurrentParseLine(")
  extract("${parser}" "${signature}" part)
  if(signature STREQUAL "char *COM_ParseExt(")
    string(REPLACE "char *COM_ParseExt(" "char *RealCOM_ParseExt(" part "${part}")
  endif()
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/parser.h" "${part}\n")
endforeach()
file(READ "${ROOT}/shared/qcommon/q_string.c" strings)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/engine_strings.h" "")
foreach(signature "int Q_stricmpn (" "int Q_stricmp (" "void Q_strncpyz(")
  extract("${strings}" "${signature}" part)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/engine_strings.h" "${part}\n")
endforeach()
file(READ "${ROOT}/codemp/rd-vulkan/tr_shader.cpp" shader)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/shader.h" "")
foreach(signature "static long generateHashValue(" "static int COM_CompressShader(")
  extract("${shader}" "${signature}" part)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/shader.h" "${part}\n")
endforeach()
string(FIND "${shader}" "#define\tMAX_SHADER_FILES" start)
string(FIND "${shader}" "shader_t *R_CreateShaderFromTextureBundle(" end)
math(EXPR length "${end}-${start}")
string(SUBSTRING "${shader}" ${start} ${length} catalogue)
file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/shader.h" "${catalogue}\n")
extract("${shader}" "static const char *FindShaderInShaderText(" find_shader)
file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/shader.h" "${find_shader}\n")
add_executable(shader_text_cache check.cpp)
target_include_directories(shader_text_cache PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
  target_compile_definitions(shader_text_cache PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
enable_testing()
add_test(NAME shader_text_cache COMMAND shader_text_cache)
