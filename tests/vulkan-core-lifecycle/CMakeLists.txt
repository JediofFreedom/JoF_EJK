function(extract file signature output)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${file}")
  file(READ "${file}" source)
  string(FIND "${source}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing production function ${signature}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} body)
  set(${output} "${body}" PARENT_SCOPE)
endfunction()

endif()
foreach(signature "qboolean vk_begin_frame(" "void vk_end_frame(" "void vk_present_frame(" "static qboolean is_bgr(" "void vk_read_pixels(")
  if(signature STREQUAL "qboolean vk_begin_frame(")
    file(READ "${CORE_SOURCE_DIR}/vk_frame.cpp" frame_source)
    string(FIND "${frame_source}" "qboolean vk_begin_frame(" modern)
    if(modern LESS 0)
      extract("${CORE_SOURCE_DIR}/vk_frame.cpp" "void vk_begin_frame(" body)
      string(REPLACE "void vk_begin_frame(" "qboolean vk_begin_frame(" body "${body}")
      string(REPLACE "return;" "return qtrue;" body "${body}")
      string(REGEX REPLACE "\n}$" "\nreturn qtrue;\n}" body "${body}")
    else()
      extract("${CORE_SOURCE_DIR}/vk_frame.cpp" "${signature}" body)
    endif()
  else()
    extract("${CORE_SOURCE_DIR}/vk_frame.cpp" "${signature}" body)
  endif()
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/frame.h" "${body}\n")
endforeach()
