cmake_minimum_required(VERSION 3.19)
project(renderer_mark_surfaces CXX)
set(CMAKE_CXX_STANDARD 11)
get_filename_component(default_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(EJK_SOURCE_DIR "${default_root}" CACHE PATH "Engine source checkout")
set(MARKS_SOURCE_DIR "${EJK_SOURCE_DIR}/codemp" CACHE PATH "Production renderer sources; may select an older source tree")
function(extract path signature output)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
  file(READ "${path}" source)
  string(FIND "${source}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing actual production definition: ${signature}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} part)
  set(${output} "${part}" PARENT_SCOPE)
endfunction()
foreach(renderer vulkan vanilla)
  extract("${MARKS_SOURCE_DIR}/rd-${renderer}/tr_marks.cpp" "void R_BoxSurfaces_r(" function)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${renderer}.h" "${function}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/primitives.h" "")
foreach(signature "int BoxOnPlaneSide(" "float DotProduct( " "void SetPlaneSignbits( " "int\tPlaneTypeForNormal( ")
  extract("${EJK_SOURCE_DIR}/shared/qcommon/q_math.c" "${signature}" function)
  string(REPLACE "int BoxOnPlaneSide(" "int MathBoxOnPlaneSide(" function "${function}")
  string(REPLACE "float DotProduct(" "float MathDotProduct(" function "${function}")
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/primitives.h" "${function}\n")
endforeach()
extract("${EJK_SOURCE_DIR}/codemp/rd-vulkan/tr_bsp.cpp" "float R_ClampDenorm(" function)
file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/primitives.h" "${function}\n")

file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/bsp_types.h" "")
file(READ "${EJK_SOURCE_DIR}/codemp/qcommon/qfiles.h" types)
foreach(pair "lump_s|lump_t" "dheader_s|dheader_t" "dshader_s|dshader_t" "dplane_s|dplane_t" "dnode_s|dnode_t" "dleaf_s|dleaf_t" "mapVert_s|mapVert_t" "dsurface_s|dsurface_t")
  string(REPLACE "|" ";" pair "${pair}")
  list(GET pair 0 tag)
  list(GET pair 1 name)
  string(FIND "${types}" "typedef struct ${tag} {" start)
  string(FIND "${types}" "} ${name};" end)
  if(start LESS 0 OR end LESS start)
    message(FATAL_ERROR "Missing actual BSP type ${name}")
  endif()
  string(LENGTH "} ${name};" suffix)
  math(EXPR length "${end}-${start}+${suffix}")
  string(SUBSTRING "${types}" ${start} ${length} type)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/bsp_types.h" "${type}\n")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${EJK_SOURCE_DIR}/codemp/qcommon/qfiles.h")
add_executable(renderer_mark_surfaces check.cpp)
target_include_directories(renderer_mark_surfaces PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${EJK_SOURCE_DIR}/codemp")
if(MSVC)
  target_compile_options(renderer_mark_surfaces PRIVATE /EHsc)
endif()
enable_testing()
add_test(NAME renderer_mark_surfaces COMMAND renderer_mark_surfaces)
