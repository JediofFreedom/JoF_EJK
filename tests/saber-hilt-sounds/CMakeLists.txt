function(extract path first last)
  set(source_path "${CMAKE_CURRENT_SOURCE_DIR}/../../${path}")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source_path}")
  file(READ "${source_path}" source)
  string(FIND "${source}" "${first}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Cannot extract ${first}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 source)
  string(FIND "${source}" "${last}" end)
  if(end LESS 0)
    message(FATAL_ERROR "Cannot extract ${first}")
  endif()
  string(SUBSTRING "${source}" 0 ${end} part)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/actual.h" "${part}\n")
endfunction()
