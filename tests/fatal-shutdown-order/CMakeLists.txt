cmake_minimum_required(VERSION 3.19)
project(fatal_shutdown_order CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(default_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(EJK_SOURCE_DIR "${default_root}" CACHE PATH "Engine checkout")
set(COMMON_SOURCE "${EJK_SOURCE_DIR}/codemp/qcommon/common.cpp" CACHE FILEPATH "Production common.cpp; may select pre-fix snapshot")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${COMMON_SOURCE}")
file(READ "${COMMON_SOURCE}" common_source)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/actual_error.h" "")
foreach(signature "void NORETURN QDECL Com_Error(" "void Com_Quit_f(")
  string(FIND "${common_source}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing actual production function ${signature}")
  endif()
  string(SUBSTRING "${common_source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  if(end LESS 0)
    message(FATAL_ERROR "Missing production function terminator")
  endif()
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} body)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/actual_error.h" "${body}\n")
endforeach()
add_executable(fatal_shutdown_order check.cpp)
target_include_directories(fatal_shutdown_order PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
  target_compile_options(fatal_shutdown_order PRIVATE /EHsc)
endif()
enable_testing()
foreach(part fatal buildscript dedicated recoverable recursive)
  add_test(NAME fatal_shutdown_${part} COMMAND fatal_shutdown_order ${part})
endforeach()
