function(extract path signature output count_calls)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/${path}")
  file(READ "${ROOT}/${path}" source)
  if(REFERENCE_REF AND output STREQUAL "subject.h")
    execute_process(COMMAND git show "${REFERENCE_REF}:${path}" WORKING_DIRECTORY "${ROOT}"
      OUTPUT_VARIABLE source COMMAND_ERROR_IS_FATAL ANY)
  endif()
  string(FIND "${source}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing production function ${signature}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} body)
  if(count_calls)
    string(FIND "${body}" "{" brace)
    math(EXPR brace "${brace}+1")
    string(SUBSTRING "${body}" 0 ${brace} prefix)
    string(SUBSTRING "${body}" ${brace} -1 suffix)
    set(body "${prefix}\n++classifierCalls;${suffix}")
  endif()
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/${output}" "${body}\n")
endfunction()
