cmake_minimum_required(VERSION 3.19)
project(roff_client_notes C CXX)
set(CMAKE_CXX_STANDARD 11)
include("${CMAKE_CURRENT_SOURCE_DIR}/../TestPlatform.cmake")
get_filename_component(repo "../.." ABSOLUTE)
set(REFERENCE_REF "" CACHE STRING "Optional git ref to reproduce client note parsing failures")
if(REFERENCE_REF)
  execute_process(COMMAND git show "${REFERENCE_REF}:codemp/cgame/cg_ents.c"
    WORKING_DIRECTORY "${repo}" OUTPUT_VARIABLE source COMMAND_ERROR_IS_FATAL ANY)
else()
  file(READ "${repo}/codemp/cgame/cg_ents.c" source)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${repo}/codemp/cgame/cg_ents.c")
endif()
string(FIND "${source}" "void CG_ROFF_NotetrackCallback(" start)
if(start LESS 0)
  message(FATAL_ERROR "Missing CG_ROFF_NotetrackCallback")
endif()
string(SUBSTRING "${source}" ${start} -1 tail)
string(FIND "${tail}" "\n}" end)
math(EXPR length "${end}+2")
string(SUBSTRING "${tail}" 0 ${length} callback)
# Poison the unused local string tail in sanitizer fixtures. A past-NUL read
# otherwise consumes uninitialized bytes inside the stack array silently.
set(marker "\t\taddlArg[r] = '\\0';")
string(FIND "${callback}" "${marker}" marker_start)
if(marker_start LESS 0)
  message(FATAL_ERROR "Missing additional-argument terminator fixture hook")
endif()
string(REPLACE "${marker}" "${marker}\n\t\tFixtureMarkUnused(addlArg + r + 1, sizeof(addlArg) - r - 1);" callback "${callback}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/client_note.h" "${callback}\n")
configure_file("${repo}/codemp/versionfile/jofversion.h.in" jofversion.h @ONLY)
add_executable(roff_client_notes check.cpp "${repo}/shared/qcommon/q_math.c")
ejk_test_platform(roff_client_notes)
target_compile_definitions(roff_client_notes PRIVATE _CGAME)
target_include_directories(roff_client_notes PRIVATE "${CMAKE_CURRENT_BINARY_DIR}"
  "${repo}/codemp" "${repo}/shared")
if(MSVC)
  target_compile_options(roff_client_notes PRIVATE /EHsc)
  target_compile_definitions(roff_client_notes PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_link_libraries(roff_client_notes PRIVATE m)
endif()
enable_testing()
foreach(case valid boundaries oversize_type oversize_argument oversize_additional
    oversize_offset oversize_angle truncated_offsets truncated_angles)
  add_test(NAME roff_client_${case} COMMAND roff_client_notes ${case})
endforeach()
