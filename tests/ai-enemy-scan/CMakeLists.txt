cmake_minimum_required(VERSION 3.16)
project(ai_enemy_scan CXX)
set(CMAKE_CXX_STANDARD 11)
if(NOT ROOT)
 get_filename_component(ROOT "../.." ABSOLUTE)
endif()
set(AI_SOURCE "${ROOT}/codemp/game/ai_main.c" CACHE FILEPATH "Actual AI source, optionally pre-optimization")
function(extract path signature output)
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
 file(READ "${path}" source)
 string(FIND "${source}" "${signature}" start)
 if(start LESS 0)
  message(FATAL_ERROR "Missing production function ${signature}")
 endif()
 string(SUBSTRING "${source}" ${start} -1 tail)
 string(FIND "${tail}" "\n}" end)
 if(end LESS 0)
  message(FATAL_ERROR "Missing production function end ${signature}")
 endif()
 math(EXPR length "${end}+2")
 string(SUBSTRING "${tail}" 0 ${length} part)
 file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/${output}.h" "${part}\n")
endfunction()
foreach(header primitives services subject)
 file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${header}.h" "")
endforeach()
file(STRINGS "${ROOT}/codemp/qcommon/q_shared.h" capacity REGEX "^#define[ \t]+MAX_CLIENTS[ \t]")
if(NOT capacity)
 message(FATAL_ERROR "Missing production MAX_CLIENTS")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/capacity.h" "${capacity}\n")
extract("${ROOT}/shared/qcommon/q_math.c" "float AngleMod(float a)" primitives)
extract("${ROOT}/shared/qcommon/q_math.c" "void vectoangles(" primitives)
extract("${ROOT}/codemp/game/ai_main.c" "int InFieldOfVision(" primitives)
foreach(signature "int PassStandardEnemyChecks(" "int PassLovedOneCheck(bot_state_t *bs, gentity_t *ent)\n{" "qboolean BotPVSCheck(" "int OrgVisible(" "int BotCanHear(")
 extract("${ROOT}/codemp/game/ai_main.c" "${signature}" services)
endforeach()
extract("${AI_SOURCE}" "int ScanForEnemies(" subject)
add_executable(ai_enemy_scan "${ROOT}/tests/ai-enemy-scan/check.cpp")
target_include_directories(ai_enemy_scan PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${ROOT}/tests/ai-enemy-scan")
if(MSVC)
 target_compile_options(ai_enemy_scan PRIVATE /EHsc)
endif()
enable_testing()
foreach(case ordered reverse new ties jedi current-jedi health hysteresis mindtrick fov degenerate last-index invalid randomized work)
 add_test(NAME ai_scan_${case} COMMAND ai_enemy_scan ${case})
endforeach()
