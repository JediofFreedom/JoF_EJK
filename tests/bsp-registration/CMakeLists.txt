cmake_minimum_required(VERSION 3.16)
project(bsp_registration CXX)
set(CMAKE_CXX_STANDARD 11)
get_filename_component(ROOT "../.." ABSOLUTE)
set(BSP_SOURCE_ROOT "${ROOT}/codemp" CACHE PATH "Desktop BSP registration sources; may select a pre-fix tree")
set(CG_VIEW_SOURCE "${ROOT}/codemp/cgame/cg_view.c" CACHE FILEPATH "Actual debug testmodel source")
function(extract path signature output)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
  file(READ "${path}" source)
  string(FIND "${source}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing ${signature} in ${path}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} function)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/${output}.h" "${function}\n")
endfunction()
foreach(output vulkan vanilla rend2 debug)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}.h" "")
endforeach()
extract("${BSP_SOURCE_ROOT}/rd-vulkan/tr_model.cpp" "static qhandle_t RE_RegisterBSP(" vulkan)
set(vanilla "${BSP_SOURCE_ROOT}/rd-vanilla/tr_model.cpp")
file(READ "${vanilla}" source)
string(FIND "${source}" "static qhandle_t RE_RegisterBSP(" helper)
if(helper LESS 0)
  # Before the fix vanilla's identical operation lives inline in the model
  # registrar. Extract that actual branch to exercise its original behavior.
  string(FIND "${source}" "\tif (name[0] == '#')" start)
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n\t}" end)
  math(EXPR length "${end}+3")
  string(SUBSTRING "${tail}" 0 ${length} branch)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/vanilla.h" "static qhandle_t RE_RegisterBSP(const char *name) {\nint hash; modelHash_t *mh;\n${branch}\nreturn 0;\n}\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${vanilla}")
else()
  extract("${vanilla}" "static qhandle_t RE_RegisterBSP(" vanilla)
endif()
# Rend2 already bounds its distinct instances in R_LoadBSP. Execute that real
# allocation prefix; the rest of its disk parsing is irrelevant to capacity.
set(rend2_bsp "${ROOT}/shared/rd-rend2/tr_bsp.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${rend2_bsp}")
file(READ "${rend2_bsp}" source)
string(FIND "${source}" "world_t *R_LoadBSP(" start)
string(SUBSTRING "${source}" ${start} -1 tail)
string(FIND "${tail}" "\n\t// load it" length)
string(SUBSTRING "${tail}" 0 ${length} prefix)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/rend2.h" "${prefix}\nreturn worldData;\n}\n")
extract("${ROOT}/shared/rd-rend2/tr_model.cpp" "static qhandle_t RE_RegisterBSP(" rend2)
extract("${CG_VIEW_SOURCE}" "void CG_TestModel_f (" debug)
extract("${CG_VIEW_SOURCE}" "static void CG_AddTestModel (" debug)
extract("${ROOT}/codemp/cgame/cg_main.c" "void CG_Init_CG(" debug)
file(STRINGS "${ROOT}/codemp/qcommon/q_shared.h" capacity REGEX "^#define[ \t]+MAX_SUB_BSP[ \t]")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/capacity.h" "${capacity}\n")
# Keeping an already registered debug BSP handle relies on cgame state being
# reset whenever the engine destroys/recreates the renderer.
file(READ "${ROOT}/codemp/cgame/cg_main.c" cg_source)
string(FIND "${cg_source}" "void CG_Init( int serverMessageNum, int serverCommandSequence, int clientNum )\n{" cg_init)
string(SUBSTRING "${cg_source}" ${cg_init} -1 cg_tail)
string(FIND "${cg_tail}" "\n\tCG_Init_CG();" cg_reset)
if(cg_reset LESS 0)
  message(FATAL_ERROR "Debug BSP handles require CG_Init to reset cg")
endif()
file(READ "${ROOT}/codemp/client/cl_main.cpp" client)
string(FIND "${client}" "void CL_Vid_Restart_f(" restart)
string(SUBSTRING "${client}" ${restart} -1 restart_tail)
string(FIND "${restart_tail}" "\n}" restart_end)
string(SUBSTRING "${restart_tail}" 0 ${restart_end} restart_body)
string(FIND "${restart_body}" "CL_ShutdownCGame();" shutdown)
string(FIND "${restart_body}" "CL_InitRef();" renderer)
string(FIND "${restart_body}" "CL_InitCGame();" init)
if(shutdown LESS 0 OR renderer LESS shutdown OR init LESS renderer)
  message(FATAL_ERROR "Renderer restarts must reinitialize cgame and its debug model handles")
endif()
foreach(path codemp/cgame/cg_main.c codemp/client/cl_main.cpp)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/${path}")
endforeach()
add_executable(bsp_registration check.cpp)
target_include_directories(bsp_registration PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
  target_compile_options(bsp_registration PRIVATE /EHsc)
endif()
enable_testing()
add_test(NAME bsp_registration COMMAND bsp_registration)
add_test(NAME bsp_vanilla_capacity COMMAND bsp_registration vanilla)
add_test(NAME debug_bsp_testmodel COMMAND bsp_registration debug)
