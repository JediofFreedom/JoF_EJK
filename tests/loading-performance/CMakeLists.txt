# Compile the production functions, with only their engine services stubbed.
function(read_source path out)
  if(REFERENCE_REF)
    execute_process(COMMAND git show "${REFERENCE_REF}:${path}" WORKING_DIRECTORY "${ROOT}"
      OUTPUT_VARIABLE source COMMAND_ERROR_IS_FATAL ANY)
  else()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/${path}")
    file(READ "${ROOT}/${path}" source)
  endif()
  set(${out} "${source}" PARENT_SCOPE)
endfunction()

function(extract_function source signature out)
  string(FIND "${source}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing ${signature}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} result)
  set(${out} "${result}" PARENT_SCOPE)
endfunction()
