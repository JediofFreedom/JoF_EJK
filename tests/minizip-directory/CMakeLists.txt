cmake_minimum_required(VERSION 3.19)
project(minizip_directory C CXX)
set(CMAKE_CXX_STANDARD 11)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(REFERENCE_REF "" CACHE STRING "Optional full minizip source revision; default uses frozen original scalar readers")
if(REFERENCE_REF)
  execute_process(COMMAND git show "${REFERENCE_REF}:lib/minizip/unzip.c"
    WORKING_DIRECTORY "${ROOT}" OUTPUT_VARIABLE reference_source COMMAND_ERROR_IS_FATAL ANY)
else()
  # Keep the remainder of production minizip identical in both executables.
  # The original scalar readers are checked in so commits and shallow clones
  # cannot silently turn this into optimized-versus-optimized comparison.
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${ROOT}/lib/minizip/unzip.c" "${CMAKE_CURRENT_SOURCE_DIR}/reference_readers.h")
  file(READ "${ROOT}/lib/minizip/unzip.c" production_source)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/reference_readers.h" reference_readers)
  string(FIND "${production_source}" "/* Read a complete little-endian field in one callback on the common path." start)
  string(FIND "${production_source}" "/* My own strcmpi / strcasecmp */" end)
  if(start LESS 0 OR end LESS_EQUAL start)
    message(FATAL_ERROR "Cannot find production scalar-reader boundaries; update the comparison fixture")
  endif()
  string(SUBSTRING "${production_source}" 0 ${start} before_readers)
  string(SUBSTRING "${production_source}" ${end} -1 after_readers)
  set(reference_source "${before_readers}${reference_readers}${after_readers}")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/reference_unzip.c" "${reference_source}")
set(ZLIB_SRCS adler32.c compress.c crc32.c deflate.c infback.c inffast.c inflate.c inftrees.c trees.c uncompr.c zutil.c)
add_library(test_zlib STATIC)
foreach(src IN LISTS ZLIB_SRCS)
  target_sources(test_zlib PRIVATE "${ROOT}/lib/zlib/${src}")
endforeach()
target_include_directories(test_zlib PUBLIC "${ROOT}/lib/zlib" "${ROOT}/lib/zlib/include")
foreach(variant reference candidate)
  if(variant STREQUAL "reference")
    set(UNZIP_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/reference_unzip.c")
  else()
    get_filename_component(UNZIP_SOURCE "${ROOT}/lib/minizip/unzip.c" ABSOLUTE)
  endif()
  configure_file(wrapper.c.in "${CMAKE_CURRENT_BINARY_DIR}/${variant}_wrapper.c" @ONLY)
  add_executable(minizip_${variant} check.cpp "${CMAKE_CURRENT_BINARY_DIR}/${variant}_wrapper.c" "${ROOT}/lib/minizip/ioapi.c")
  target_include_directories(minizip_${variant} PRIVATE "${ROOT}/lib/minizip/include" "${ROOT}/lib/minizip/include/minizip")
  target_link_libraries(minizip_${variant} PRIVATE test_zlib)
  if(MSVC)
    target_compile_definitions(minizip_${variant} PRIVATE _CRT_SECURE_NO_WARNINGS)
  endif()
endforeach()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/fixtures.py" "${CMAKE_CURRENT_BINARY_DIR}/fixtures" COMMAND_ERROR_IS_FATAL ANY)
enable_testing()
add_test(NAME minizip_compatibility COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/compare.py"
  "$<TARGET_FILE:minizip_reference>" "$<TARGET_FILE:minizip_candidate>" "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
