enable_testing()
file(READ "${ROOT}/codemp/rd-vulkan/tr_local.h" local_header)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/codemp/rd-vulkan/tr_local.h")
string(FIND "${local_header}" "static QINLINE unsigned int log2pad(" log_start)
string(SUBSTRING "${local_header}" ${log_start} -1 log_tail)

math(EXPR log_length "${log_end}+2")
string(SUBSTRING "${log_tail}" 0 ${log_length} log_code)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/log2pad.h" "${log_code}\n")
function(extract path signature output optional)
  file(READ "${PRODUCTION_ROOT}/${path}" source)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PRODUCTION_ROOT}/${path}")
  string(FIND "${source}" "${signature}" start)
  if(start LESS 0)
    if(optional)
      return()
    endif()
    message(FATAL_ERROR "Missing ${signature}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} function)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/${output}.h" "${function}\n")
endfunction()
