cmake_minimum_required(VERSION 3.16)
project(staff_shutdown C)

set(source_file "${CMAKE_CURRENT_SOURCE_DIR}/../../codemp/cgame/cg_players.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source_file}")
file(READ "${source_file}" source)
string(FIND "${source}" "static void CG_StaffSwapShutdownSound(" start)
string(FIND "${source}" "//Whether this client's staff" end)
if(start LESS 0 OR end LESS start)
  message(FATAL_ERROR "Cannot extract staff shutdown sound function")
endif()
math(EXPR length "${end}-${start}")
string(SUBSTRING "${source}" ${start} ${length} function)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/actual.h" "${function}")

add_executable(staff_shutdown check.c)
target_include_directories(staff_shutdown PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
enable_testing()
add_test(NAME staff_shutdown COMMAND staff_shutdown)
