cmake_minimum_required(VERSION 3.16)
project(vulkan_frontend_shader_token CXX)
set(CMAKE_CXX_STANDARD 11)
set(VULKAN_SOURCE_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../codemp/rd-vulkan" CACHE PATH "Production renderer source")
set(path "${VULKAN_SOURCE_DIRECTORY}/tr_shader.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
file(READ "${path}" source)
string(FIND "${source}" "else if (!Q_stricmpn(token, \"ss\", 2))" start)
if(start LESS 0)
  message(FATAL_ERROR "Missing actual ParseStage surface-sprite optional dispatch")
endif()
string(SUBSTRING "${source}" ${start} -1 tail)
string(FIND "${tail}" "{" open)
math(EXPR open "${open}+1")
string(FIND "${tail}" "\n\t\telse\n" end)
math(EXPR length "${end}-${open}")
string(SUBSTRING "${tail}" ${open} ${length} branch)
# Remove only the dispatch branch's closing brace. The do/while supplies the
# surrounding parser loop for its production continue; all branch statements
# (including the fixed-size copy and stage counters) compile verbatim.
string(REGEX REPLACE "\n\t\t}[\r\n\t ]*$" "" branch "${branch}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/shader-token.h" "static void parseOptional(shaderStage_t *stage,const char **text,char *token) { do {\n${branch}\n} while (false); }\n")
add_executable(shader_token check.cpp)
target_include_directories(shader_token PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
enable_testing()
add_test(NAME shader_token COMMAND shader_token)
