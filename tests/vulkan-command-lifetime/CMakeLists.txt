cmake_minimum_required(VERSION 3.19)
project(vulkan_command_lifetime CXX)
set(CMAKE_CXX_STANDARD 11)
get_filename_component(default_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(EJK_SOURCE_DIR "${default_root}" CACHE PATH "Engine source checkout")
set(COMMAND_SOURCE_DIR "${EJK_SOURCE_DIR}/codemp/rd-vulkan" CACHE PATH "Actual command/frame sources; may select pre-fix snapshot")
function(extract file signature output)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${file}")
  file(READ "${file}" source)
  string(FIND "${source}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing actual production function ${signature}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} body)
  set(${output} "${body}" PARENT_SCOPE)
endfunction()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/commands.h" "")
foreach(signature "VkCommandBuffer vk_begin_command_buffer(" "void vk_end_command_buffer(")
  extract("${COMMAND_SOURCE_DIR}/vk_cmd.cpp" "${signature}" body)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/commands.h" "${body}\n")
endforeach()
foreach(signature "static void vk_frame_check_result(" "void vk_queue_wait_idle(")
  extract("${COMMAND_SOURCE_DIR}/vk_frame.cpp" "${signature}" body)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/commands.h" "${body}\n")
endforeach()
add_executable(vulkan_command_lifetime check.cpp)
target_include_directories(vulkan_command_lifetime PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${EJK_SOURCE_DIR}/codemp/rd-vulkan")
if(MSVC)
  target_compile_options(vulkan_command_lifetime PRIVATE /EHsc)
endif()
