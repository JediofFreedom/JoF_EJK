cmake_minimum_required(VERSION 3.16)
project(client_shutdown CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(CL_MAIN_SOURCE "${repo}/codemp/client/cl_main.cpp" CACHE FILEPATH "Client source, optionally a pre-fix snapshot")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CL_MAIN_SOURCE}")
file(READ "${CL_MAIN_SOURCE}" source)
string(FIND "${source}" "void CL_Shutdown( void ) {" start)
if(start LESS 0)
  message(FATAL_ERROR "Cannot find production CL_Shutdown")
endif()
string(SUBSTRING "${source}" ${start} -1 tail)
string(FIND "${tail}" "\n}" end)
if(end LESS 0)
  message(FATAL_ERROR "Cannot find end of production CL_Shutdown")
endif()
math(EXPR length "${end} + 2")
string(SUBSTRING "${tail}" 0 ${length} body)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/actual_shutdown.h" "${body}\n")

enable_testing()
foreach(target client_shutdown client_shutdown_no_discord)
  add_executable(${target} check.cpp)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  if(MSVC)
    target_compile_options(${target} PRIVATE /EHsc)
  endif()
endforeach()
target_compile_definitions(client_shutdown PRIVATE DISCORD FINAL_BUILD)
foreach(part early disabled enabled disabled_initialized null_initialized repeated recursive)
  add_test(NAME client_shutdown_${part} COMMAND client_shutdown ${part})
endforeach()
add_test(NAME client_shutdown_no_discord COMMAND client_shutdown_no_discord early)
