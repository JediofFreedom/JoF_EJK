cmake_minimum_required(VERSION 3.16)
project(icarus_task_parameters CXX)
set(CMAKE_CXX_STANDARD 11)
get_filename_component(ROOT "../.." ABSOLUTE)
set(ICARUS_TASK_SOURCE "${ROOT}/codemp/icarus/TaskManager.cpp" CACHE FILEPATH "Production task manager source (can select a baseline fixture)")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ICARUS_TASK_SOURCE}")
file(READ "${ICARUS_TASK_SOURCE}" source)
string(REGEX MATCH "#define ICARUS_VALIDATE[^\n]+" validation "${source}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/helpers.h" "${validation}\n")
foreach(signature "inline bool CTaskManager::Check(" "int CTaskManager::GetFloat(" "int CTaskManager::GetVector(" "int CTaskManager::Get(" "int CTaskManager::Move(")
  string(FIND "${source}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing ${signature}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} function)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/helpers.h" "${function}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/tokens.h" "")
foreach(pair "tokenizer.h|TK_EOF = -1" "interpreter.h|TK_BLOCK_START = TK_USERDEF" "interpreter.h|ID_AFFECT = NUM_USER_TOKENS" "taskmanager.h|TASK_OK,")
  string(REPLACE "|" ";" fields "${pair}")
  list(GET fields 0 header)
  list(GET fields 1 first)
  set(path "${ROOT}/codemp/icarus/${header}")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
  file(READ "${path}" types)
  string(REGEX MATCH "enum[ \t\r\n]*\\{[ \t\r\n]*${first}[^}]+\\};" enum "${types}")
  if(NOT enum)
    message(FATAL_ERROR "Missing production enum ${first}")
  endif()
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/tokens.h" "${enum}\n")
endforeach()
add_executable(icarus_task_parameters check.cpp)
target_include_directories(icarus_task_parameters PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
# The release engine reports malformed script parameters and returns failure;
# assertions in these paths are intentionally disabled in this regression.
target_compile_definitions(icarus_task_parameters PRIVATE NDEBUG)
if(MSVC)
  target_compile_options(icarus_task_parameters PRIVATE /EHsc)
endif()
enable_testing()
add_test(NAME icarus_task_parameters COMMAND icarus_task_parameters)
