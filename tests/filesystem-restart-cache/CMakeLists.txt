cmake_minimum_required(VERSION 3.19)
project(filesystem_restart_cache C CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
get_filename_component(ROOT "../.." ABSOLUTE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/codemp/qcommon/files.cpp")
file(READ "${ROOT}/codemp/qcommon/files.cpp" source)
function(extract signature output)
  string(FIND "${source}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing production function ${signature}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} part)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/${output}.h" "${part}\n")
endfunction()
string(FIND "${source}" "typedef struct fileInPack_s" start)
string(FIND "${source}" "static char\t\t\tfs_gamedir" end)
math(EXPR length "${end}-${start}")
string(SUBSTRING "${source}" ${start} ${length} types)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/types.h" "${types}\n")
foreach(output load mount read restart)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}.h" "")
endforeach()
foreach(signature "static long FS_HashFileName(" "static pack_t *FS_LoadZipFile(" "void FS_FreePak(")
  extract("${signature}" load)
endforeach()
foreach(signature "qboolean FS_FilenameCompare(" "static int QDECL paksort(" "static void FS_AddGameDirectory(" "static void FS_ReorderPurePaks()" "const char *FS_CmdlineGame(" "static std::string FS_RestartConfiguration(" "void FS_Startup(")
  extract("${signature}" mount)
endforeach()
foreach(signature "static long FS_FOpenFileReadInternal(" "long FS_FOpenFileRead(" "qboolean FS_GetReadFileIdentity(")
  extract("${signature}" read)
endforeach()
foreach(signature "void FS_ClearPakReferences(" "void FS_Shutdown(" "static void FS_Restart( int checksumFeed, qboolean reuseMetadata )" "void FS_Restart( int checksumFeed )" "qboolean FS_ConditionalRestart(")
  extract("${signature}" restart)
endforeach()
file(READ "${ROOT}/codemp/qcommon/md4.cpp" checksum)
string(FIND "${checksum}" "typedef struct mdfour_s" start)
string(SUBSTRING "${checksum}" ${start} -1 checksum)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/checksum.h" "${checksum}\n")

  "${ROOT}/lib/minizip/include" "${ROOT}/lib/minizip/include/minizip" "${ROOT}/lib/zlib" "${ROOT}/lib/zlib/include")
