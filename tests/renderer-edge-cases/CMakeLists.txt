cmake_minimum_required(VERSION 3.19)
project(renderer_edge_cases CXX)
set(CMAKE_CXX_STANDARD 11)
get_filename_component(ROOT "../.." ABSOLUTE)
enable_testing()
set(source_path "${ROOT}/shared/rd-rend2/tr_image_stb.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source_path}")
file(READ "${source_path}" source)
string(FIND "${source}" "void R_LoadHDRImage(" start)
string(SUBSTRING "${source}" ${start} -1 reader)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/hdr.h" "${reader}")
add_executable(renderer_edge_cases check.cpp)
target_include_directories(renderer_edge_cases PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${ROOT}/shared/qcommon" "${ROOT}/shared/rd-rend2")
if(MSVC)
  target_compile_definitions(renderer_edge_cases PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_test(NAME renderer_edge_cases COMMAND renderer_edge_cases)
