cmake_minimum_required(VERSION 3.19)
project(vulkan_client_scaling CXX)
set(CMAKE_CXX_STANDARD 11)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(EJK_SOURCE_DIR "${root}" CACHE PATH "Production checkout")
function(extract file signature)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${file}")
  file(READ "${file}" source)
  string(FIND "${source}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing production function ${signature}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} body)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/production.h" "${body}\n")
endfunction()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/production.h" "")
set(rd "${EJK_SOURCE_DIR}/codemp/rd-vulkan")
extract("${rd}/tr_model.cpp" "void RE_BeginRegistration(")
extract("${rd}/tr_init.cpp" "static void GetRealRes(")
foreach(name StretchPic RotatePic RotatePic2)
  extract("${rd}/tr_cmds.cpp" "void RE_${name} (")
endforeach()
# Compile the actual scene-coordinate conversion, without a simulated world.
file(READ "${rd}/tr_scene.cpp" scene)
string(FIND "${scene}" "\n\ttr.refdef.x =" start)
string(FIND "${scene}" "\n\ttr.refdef.fov_x =" end)
math(EXPR length "${end}-${start}")
string(SUBSTRING "${scene}" ${start} ${length} body)
file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/production.h" "void SceneCoordinates(const refdef_t *fd) {${body}\n}\n")
add_executable(vulkan_client_scaling check.cpp)
target_include_directories(vulkan_client_scaling PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
enable_testing()
add_test(NAME vulkan_client_scaling COMMAND vulkan_client_scaling)
