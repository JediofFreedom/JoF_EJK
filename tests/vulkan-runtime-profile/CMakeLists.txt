cmake_minimum_required(VERSION 3.16)
project(vulkan_runtime_profile CXX)
set(CMAKE_CXX_STANDARD 11)
find_package(Threads REQUIRED)
add_executable(vulkan_runtime_profile check.cpp)
target_include_directories(vulkan_runtime_profile PRIVATE ../../codemp/rd-vulkan)
target_link_libraries(vulkan_runtime_profile PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(vulkan_runtime_profile PRIVATE /EHsc)
endif()
enable_testing()
add_test(NAME vulkan_runtime_profile COMMAND vulkan_runtime_profile)
