cmake_minimum_required(VERSION 3.16)
project(vehicle_lifecycle CXX)
set(CMAKE_CXX_STANDARD 11)
if(NOT ROOT)
 get_filename_component(ROOT "../.." ABSOLUTE)
endif()
set(VEHICLE_SOURCE "${ROOT}/codemp/game/g_vehicles.c" CACHE FILEPATH "Actual vehicle source, optionally pre-fix")
set(PMOVE_SOURCE "${ROOT}/codemp/game/bg_pmove.c" CACHE FILEPATH "Actual movement source, optionally pre-fix")
set(extracted_header vehicle_functions.h)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/vehicle_functions.h" "")
function(extract path signature)
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
 file(READ "${path}" source)
 string(FIND "${source}" "${signature}" start)
 if(start LESS 0)
  message(FATAL_ERROR "Missing ${signature}")
 endif()
 string(SUBSTRING "${source}" ${start} -1 tail)
 string(FIND "${tail}" "\n}" end)
 math(EXPR length "${end}+2")
 string(SUBSTRING "${tail}" 0 ${length} function)
 file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/${extracted_header}" "${function}\n")
endfunction()
foreach(signature "qboolean ValidateBoard(" "qboolean Board(" "qboolean Eject("
                  "static void Ghost(" "static void UnGhost(")
 extract("${VEHICLE_SOURCE}" "${signature}")
endforeach()
extract("${VEHICLE_SOURCE}" "void G_EjectDroidUnit(")
extract("${VEHICLE_SOURCE}" "qboolean EjectAll(")
file(STRINGS "${VEHICLE_SOURCE}" pilot REGEX "^void SetPilot\\(")
file(STRINGS "${VEHICLE_SOURCE}" inhabited REGEX "^qboolean Inhabited\\(")
string(REPLACE "\\;" ";" pilot "${pilot}")
string(REPLACE "\\;" ";" inhabited "${inhabited}")
file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/vehicle_functions.h" "${pilot}\n${inhabited}\n")
file(READ "${VEHICLE_SOURCE}" source)
string(FIND "${source}" "// If we're not empty..." start)
string(SUBSTRING "${source}" ${start} -1 tail)
string(FIND "${tail}" "// Copy over the commands for local storage." end)
if(start LESS 0 OR end LESS 0)
 message(FATAL_ERROR "Missing actual Update passenger block")
endif()
string(SUBSTRING "${tail}" 0 ${end} block)
file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/vehicle_functions.h" "static void RunPassengerUpdate(Vehicle_t *pVeh) { int i;\n${block}\n}\n")
extract("${ROOT}/codemp/game/NPC_reactions.c" "void NPC_Use(")
extract("${ROOT}/codemp/game/g_utils.c" "gentity_t *G_Find (")
extract("${ROOT}/codemp/game/g_utils.c" "void GlobalUse(")
extract("${ROOT}/codemp/game/g_utils.c" "void G_UseTargets2(")
file(STRINGS "${ROOT}/codemp/game/bg_vehicles.h" capacity REGEX "^#define VEH_MAX_PASSENGERS")
if(NOT capacity)
 message(FATAL_ERROR "Missing production passenger capacity")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/vehicle_capacity.h" "${capacity}\n")
set(extracted_header seat_functions.h)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/seat_functions.h" "")
extract("${ROOT}/codemp/game/bg_vehicleLoad.c" "void BG_VehicleClampData(")
extract("${ROOT}/codemp/game/bg_vehicleLoad.c" "void AttachRidersGeneric(")
extract("${VEHICLE_SOURCE}" "static void AttachRiders(")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PMOVE_SOURCE}" "${ROOT}/codemp/game/bg_vehicles.h")
file(READ "${PMOVE_SOURCE}" source)
string(FIND "${source}" "//update the passengers" start)
if(start LESS 0)
 message(FATAL_ERROR "Missing actual movement passenger loop")
endif()
string(SUBSTRING "${source}" ${start} -1 tail)
string(FIND "${tail}" "#else" end)
if(end LESS 0)
 message(FATAL_ERROR "Missing movement passenger loop boundary")
endif()
string(SUBSTRING "${tail}" 0 ${end} block)
file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/seat_functions.h" "static void RunPassengerMoves(bgEntity_t *veh) { int i=0;\n${block}\n}\n")
add_executable(vehicle_lifecycle "${ROOT}/tests/vehicle-lifecycle/check.cpp")
target_compile_definitions(vehicle_lifecycle PRIVATE VEHICLE_SEAT_CONSUMERS VEHICLE_EJECT_ALL)
target_include_directories(vehicle_lifecycle PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
 target_compile_options(vehicle_lifecycle PRIVATE /EHsc)
endif()

 all-solid-live all-solid-dead all-reject-callback all-inert-callback all-mutate-next
