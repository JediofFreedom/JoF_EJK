cmake_minimum_required(VERSION 3.16)
project(vulkan_sprite_vbo_identity CXX)
set(CMAKE_CXX_STANDARD 11)
get_filename_component(ROOT "../.." ABSOLUTE)
set(VBO_SOURCE "${ROOT}/codemp/rd-vulkan/vk_vbo.cpp" CACHE FILEPATH "Actual static VBO allocator source")
set(VBO_TYPES "${ROOT}/codemp/rd-vulkan/tr_local.h" CACHE FILEPATH "Actual VBO capacity and sprite packing definitions")
foreach(path "${VBO_SOURCE}" "${VBO_TYPES}")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
endforeach()
file(READ "${VBO_SOURCE}" source)
string(FIND "${source}" "VBO_t *R_CreateVBO(" start)
if(start LESS 0)
  message(FATAL_ERROR "Missing actual static VBO allocator")
endif()
string(SUBSTRING "${source}" ${start} -1 tail)
string(FIND "${tail}" "\n}" end)
math(EXPR length "${end}+2")
string(SUBSTRING "${tail}" 0 ${length} function)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/create_vbo.h" "${function}\n")
file(STRINGS "${VBO_TYPES}" definitions REGEX "^[ \t]*#define[ \t]+(SS_(ENT|VBO|FOG)_(BITS|MASK)|SS_PACK_SURF_BITS|SS_UNPACK_(ENT|VBO|FOG)|MAX_VBOS)[ \t(]")
string(REPLACE ";" "\n" definitions "${definitions}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/sprite_bits.h" "${definitions}\n")
file(READ "${VBO_TYPES}" types)
string(REGEX MATCH "VBO_t[ \t]+\\*vbos\\[[^]]+\\]" array "${types}")
if(NOT array)
  message(FATAL_ERROR "Missing actual renderer VBO array")
endif()
string(REGEX MATCH "IBO_t[ \t]+\\*ibos\\[[^]]+\\]" index_array "${types}")
if(NOT index_array)
  message(FATAL_ERROR "Missing actual renderer IBO array")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/vbo_array.h" "${array};\n${index_array};\n")
add_executable(vulkan_sprite_vbo_identity check.cpp)
target_include_directories(vulkan_sprite_vbo_identity PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
  target_compile_options(vulkan_sprite_vbo_identity PRIVATE /EHsc)
endif()
enable_testing()
add_test(NAME vulkan_sprite_vbo_identity COMMAND vulkan_sprite_vbo_identity)
