cmake_minimum_required(VERSION 3.19)
project(filesystem_provider_index C CXX)
set(CMAKE_CXX_STANDARD 11)
enable_testing()
get_filename_component(ROOT "../.." ABSOLUTE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/codemp/qcommon/files.cpp")
file(READ "${ROOT}/codemp/qcommon/files.cpp" source)
function(extract signature output)
  string(FIND "${source}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing ${signature}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} part)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/${output}.h" "${part}\n")
endfunction()
foreach(output filesystem filesystem_hash filesystem_compare reorder)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}.h" "")
endforeach()
foreach(signature "static long FS_FOpenFileReadInternal(" "long FS_FOpenFileRead(" "qboolean FS_GetReadFileIdentity(" "int\t\tFS_FOpenFileByMode(")
  extract("${signature}" filesystem)
endforeach()
extract("static long FS_HashFileName(" filesystem_hash)
extract("qboolean FS_FilenameCompare(" filesystem_compare)
file(READ "${CMAKE_CURRENT_BINARY_DIR}/filesystem_compare.h" compare)
string(REPLACE "FS_FilenameCompare(" "FS_ProviderFixtureCompare(" compare "${compare}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/filesystem_compare.h" "${compare}")
extract("static void FS_ReorderPurePaks()" reorder)
# Lifecycle checks supplement the actual reorder/mount replacement fixtures.
foreach(signature "static void FS_AddGameDirectory(" "void FS_Shutdown(")
  string(FIND "${source}" "${signature}" start)
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  string(SUBSTRING "${tail}" 0 ${end} lifecycle)
  if(signature MATCHES "AddGameDirectory" AND NOT lifecycle MATCHES "fs_pakLookupIndex.invalidate")
    message(FATAL_ERROR "Mount changes must invalidate the provider index")
  elseif(signature MATCHES "Shutdown" AND NOT lifecycle MATCHES "fs_pakLookupIndex.reset")
    message(FATAL_ERROR "Shutdown must discard member pointers before freeing providers")
  endif()
  if(signature MATCHES "Shutdown")
    string(FIND "${lifecycle}" "fs_pakLookupIndex.reset" reset_position)
    string(FIND "${lifecycle}" "FS_FreePak" free_position)
    if(NOT reset_position LESS free_position)
      message(FATAL_ERROR "Provider snapshots must be discarded before package storage")
    endif()
  endif()
endforeach()
add_executable(filesystem_provider_index check.cpp "${ROOT}/lib/minizip/unzip.c" "${ROOT}/lib/minizip/ioapi.c")
target_include_directories(filesystem_provider_index PRIVATE "${CMAKE_CURRENT_BINARY_DIR}"
  "${ROOT}/codemp/qcommon" "${ROOT}/lib/minizip/include" "${ROOT}/lib/minizip/include/minizip"
  "${ROOT}/lib/zlib" "${ROOT}/lib/zlib/include")
foreach(src adler32.c compress.c crc32.c deflate.c infback.c inffast.c inflate.c inftrees.c trees.c uncompr.c zutil.c)
  target_sources(filesystem_provider_index PRIVATE "${ROOT}/lib/zlib/${src}")
endforeach()
if(MSVC)
  target_compile_options(filesystem_provider_index PRIVATE /EHsc)
  target_compile_definitions(filesystem_provider_index PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
find_package(Threads REQUIRED)
target_link_libraries(filesystem_provider_index PRIVATE Threads::Threads)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/fixtures.py"
  "${CMAKE_CURRENT_BINARY_DIR}" COMMAND_ERROR_IS_FATAL ANY)
add_test(NAME filesystem_provider_index COMMAND filesystem_provider_index "${CMAKE_CURRENT_BINARY_DIR}")
