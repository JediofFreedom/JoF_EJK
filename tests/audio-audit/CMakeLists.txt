cmake_minimum_required(VERSION 3.16)
project(audio_audit CXX)
set(CMAKE_CXX_STANDARD 11)
get_filename_component(ROOT "../.." ABSOLUTE)
enable_testing()

function(extract path signature)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/${path}")
  file(READ "${ROOT}/${path}" source)
  string(FIND "${source}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing production function: ${signature}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} function)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/production.h" "${function}\n")
endfunction()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/production.h" "")
extract(codemp/client/snd_mem.cpp "void ResampleSfx (")
extract(codemp/client/snd_dma.cpp "static qboolean S_ReadMusicWavHeader(")
extract(codemp/client/snd_dma.cpp "static int S_BackgroundReadSamples(")
file(READ "${ROOT}/codemp/client/snd_dma.cpp" sound_source)
foreach(required "Cvar_Get(\"s_loadProfile\", \"0\"" "Cmd_AddCommand(\"s_loadProfileReset\"" "Cmd_AddCommand(\"s_loadProfileReport\"" "Cmd_RemoveCommand(\"s_loadProfileReset\"" "Cmd_RemoveCommand(\"s_loadProfileReport\"")
  string(FIND "${sound_source}" "${required}" found)
  if(found LESS 0)
    message(FATAL_ERROR "Missing sound load profiler lifecycle action: ${required}")
  endif()
endforeach()
file(READ "${ROOT}/codemp/client/snd_local.h" types)
string(REGEX MATCH "typedef struct wavinfo_s[^}]+} wavinfo_t;" wave_type "${types}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/wave_type.h" "${wave_type}")
add_executable(audio_audit check.cpp)
target_include_directories(audio_audit PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME audio_audit COMMAND audio_audit)
