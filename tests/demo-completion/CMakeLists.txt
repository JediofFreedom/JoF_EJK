cmake_minimum_required(VERSION 3.19)
project(demo_completion CXX)
set(CMAKE_CXX_STANDARD 11)
get_filename_component(repo "../.." ABSOLUTE)
set(REFERENCE_REF "" CACHE STRING "Optional git ref to reproduce demo completion failure")
if(REFERENCE_REF)
  execute_process(COMMAND git show "${REFERENCE_REF}:codemp/client/cl_main.cpp"
    WORKING_DIRECTORY "${repo}" OUTPUT_VARIABLE source COMMAND_ERROR_IS_FATAL ANY)
else()
  file(READ "${repo}/codemp/client/cl_main.cpp" source)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${repo}/codemp/client/cl_main.cpp")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/demo_methods.h" "")
foreach(signature "void CL_Disconnect( qboolean showMainMenu )"
    "void CL_Disconnect_f( void )" "void CL_NextDemo( void )" "void CL_DemoCompleted( void )")
  string(FIND "${source}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing ${signature}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} method)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/demo_methods.h" "${method}\n")
endforeach()
add_executable(demo_completion check.cpp)
target_include_directories(demo_completion PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
  target_compile_options(demo_completion PRIVATE /EHsc)
  target_compile_definitions(demo_completion PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
enable_testing()
foreach(case nextdemo menu missing_ui zero_elapsed manual_disconnect)
  add_test(NAME demo_${case} COMMAND demo_completion ${case})
endforeach()
