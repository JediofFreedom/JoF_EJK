cmake_minimum_required(VERSION 3.16)
project(server_runtime_profile CXX)
set(CMAKE_CXX_STANDARD 11)
if(NOT ROOT)
 get_filename_component(ROOT "../.." ABSOLUTE)
endif()
function(extract path signature)
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/${path}")
 file(READ "${ROOT}/${path}" source)
 string(FIND "${source}" "${signature}" start)
 if(start LESS 0)
  message(FATAL_ERROR "Missing production wrapper ${signature}")
 endif()
 string(SUBSTRING "${source}" ${start} -1 tail)
 string(FIND "${tail}" "\n}" end)
 math(EXPR length "${end}+2")
 string(SUBSTRING "${tail}" 0 ${length} part)
 file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/wrappers.h" "${part}\n")
endfunction()
set(profile_source "${ROOT}/codemp/server/sv_runtime_profile.cpp")
set(profile_header "${ROOT}/codemp/server/sv_runtime_profile.h")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${profile_source}" "${profile_header}")
file(READ "${profile_source}" source)
# Replace only the engine umbrella include; compile the production implementation.
string(REPLACE "#include \"server.h\"" "" source "${source}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/profile_body.h" "${source}")
file(READ "${profile_header}" header)
# A deterministic chrono-compatible clock counts every production clock access.
string(REPLACE "using SVRuntimeClock = std::chrono::steady_clock;" "using SVRuntimeClock = TestClock;" header "${header}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/sv_runtime_profile.h" "${header}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/wrappers.h" "")
foreach(signature "void GVM_InitGame(" "void GVM_ShutdownGame(" "void GVM_RunFrame(" "int GVM_BotAIStartFrame(" "void SV_UnbindGame(" "void SV_RestartGame(")
 extract(codemp/server/sv_gameapi.cpp "${signature}")
endforeach()
extract(codemp/server/sv_bot.cpp "void SV_BotFrame( int time ) {")
extract(codemp/server/sv_main.cpp "void SV_Frame( int msec ) {")
extract(codemp/server/sv_game.cpp "void SV_ShutdownGameProgs( void ) {")
extract(codemp/server/sv_init.cpp "void SV_Shutdown( char *finalmsg )")
add_executable(server_runtime_profile "${ROOT}/tests/server-runtime-profile/check.cpp")
target_include_directories(server_runtime_profile PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
 target_compile_options(server_runtime_profile PRIVATE /EHsc)
 target_compile_definitions(server_runtime_profile PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
enable_testing()
foreach(case listen dedicated legacy catchup stopped paused kill no-vm no-bots no-game deferred-restart
 commands restart restart-legacy map shutdown shutdown-error reset-midframe exception disable-midframe enable-midframe cli)
 add_test(NAME server_profile_${case} COMMAND server_runtime_profile ${case})
endforeach()
