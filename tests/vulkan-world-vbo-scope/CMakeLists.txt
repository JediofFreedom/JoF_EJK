cmake_minimum_required(VERSION 3.16)
project(vulkan_world_vbo_scope CXX)
set(CMAKE_CXX_STANDARD 11)
get_filename_component(ROOT "../.." ABSOLUTE)
set(WORLD_LOADER_SOURCE "${ROOT}/codemp/rd-vulkan/tr_bsp.cpp" CACHE FILEPATH "Actual loader source; may select a pre-fix snapshot")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${WORLD_LOADER_SOURCE}")
file(READ "${WORLD_LOADER_SOURCE}" source)
string(FIND "${source}" "void RE_LoadWorldMap_Actual(" start)
if(start LESS 0)
  message(FATAL_ERROR "Missing actual Vulkan world loader")
endif()
string(SUBSTRING "${source}" ${start} -1 tail)
string(FIND "${tail}" "\n}" end)
math(EXPR length "${end}+2")
string(SUBSTRING "${tail}" 0 ${length} function)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/world_loader.h" "${function}\n")
set(WORLD_VBO_SOURCE "${ROOT}/codemp/rd-vulkan/vk_vbo.cpp" CACHE FILEPATH "Actual world VBO registration source")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${WORLD_VBO_SOURCE}")
file(READ "${WORLD_VBO_SOURCE}" vbo_source)
string(FIND "${vbo_source}" "} vboSpan_t;" span_end)
if(span_end LESS 0)
  message(FATAL_ERROR "Missing world VBO registration")
endif()
string(SUBSTRING "${vbo_source}" 0 ${span_end} before_span)
string(FIND "${before_span}" "typedef struct {" span_start REVERSE)
string(SUBSTRING "${vbo_source}" ${span_start} -1 vbo_tail)
string(FIND "${vbo_tail}" "static msurface_t *vbo_nthSurface(" nth_start)
if(nth_start LESS 0)
  message(FATAL_ERROR "Missing vbo_nthSurface")
endif()
string(SUBSTRING "${vbo_tail}" ${nth_start} -1 nth_tail)
string(FIND "${nth_tail}" "\n}" nth_end)
math(EXPR registration_length "${nth_start}+${nth_end}+2")
string(SUBSTRING "${vbo_tail}" 0 ${registration_length} registration)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/world_vbo_registration.h" "${registration}\n")
# Production RE_EndRegistration, so a skipped or misplaced R_BuildWorldVBO call fails the fixture.
# The cl_loadTimings report at its end only prints load statistics; it is cut off
# so the fixture does not have to mirror every vkLoadStats field.
set(REGISTRATION_SOURCE "${ROOT}/codemp/rd-vulkan/tr_init.cpp" CACHE FILEPATH "Actual RE_EndRegistration source")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${REGISTRATION_SOURCE}")
file(READ "${REGISTRATION_SOURCE}" init_source)
string(FIND "${init_source}" "void RE_EndRegistration( void ) {" end_start)
if(end_start LESS 0)
  message(FATAL_ERROR "Missing RE_EndRegistration")
endif()
string(SUBSTRING "${init_source}" ${end_start} -1 end_tail)
string(FIND "${end_tail}" "\n}" end_end)
math(EXPR end_length "${end_end}+2")
string(SUBSTRING "${end_tail}" 0 ${end_length} end_registration)
string(FIND "${end_registration}" "if (ri.Cvar_VariableIntegerValue(\"cl_loadTimings\"))" timings)
if(NOT timings LESS 0)
  string(SUBSTRING "${end_registration}" 0 ${timings} end_registration)
  string(APPEND end_registration "\n}")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/end_registration.h" "${end_registration}\n")
add_executable(vulkan_world_vbo_scope check.cpp)
target_include_directories(vulkan_world_vbo_scope PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
  target_compile_options(vulkan_world_vbo_scope PRIVATE /EHsc)
endif()
enable_testing()
add_test(NAME vulkan_world_vbo_scope COMMAND vulkan_world_vbo_scope)
