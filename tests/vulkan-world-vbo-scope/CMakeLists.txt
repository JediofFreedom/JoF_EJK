cmake_minimum_required(VERSION 3.16)
project(vulkan_world_vbo_scope CXX)
set(CMAKE_CXX_STANDARD 11)
get_filename_component(ROOT "../.." ABSOLUTE)
set(WORLD_LOADER_SOURCE "${ROOT}/codemp/rd-vulkan/tr_bsp.cpp" CACHE FILEPATH "Actual loader source; may select a pre-fix snapshot")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${WORLD_LOADER_SOURCE}")
file(READ "${WORLD_LOADER_SOURCE}" source)
string(FIND "${source}" "void RE_LoadWorldMap_Actual(" start)
if(start LESS 0)
  message(FATAL_ERROR "Missing actual Vulkan world loader")
endif()
string(SUBSTRING "${source}" ${start} -1 tail)
string(FIND "${tail}" "\n}" end)
math(EXPR length "${end}+2")
string(SUBSTRING "${tail}" 0 ${length} function)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/world_loader.h" "${function}\n")
add_executable(vulkan_world_vbo_scope check.cpp)
target_include_directories(vulkan_world_vbo_scope PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
  target_compile_options(vulkan_world_vbo_scope PRIVATE /EHsc)
endif()
enable_testing()
add_test(NAME vulkan_world_vbo_scope COMMAND vulkan_world_vbo_scope)
