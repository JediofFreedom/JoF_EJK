cmake_minimum_required(VERSION 3.19)
project(vulkan_resource_memory CXX)
set(CMAKE_CXX_STANDARD 11)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(PRODUCTION_ROOT "${ROOT}" CACHE PATH "Production tree or before-fix snapshot")
function(extract path signature output)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PRODUCTION_ROOT}/${path}")
  file(READ "${PRODUCTION_ROOT}/${path}" source)
  string(FIND "${source}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing ${signature}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} part)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}" "${part}\n")
endfunction()
extract(codemp/rd-vulkan/vk_shaders.cpp "static VkShaderModule SHADER_MODULE(" shader.h)
extract(codemp/rd-vulkan/vk_image.cpp "static void allocate_and_bind_image_memory(" image.h)
extract(codemp/rd-vulkan/vk_pipeline_cache.cpp "void vk_create_pipeline_cache(" cache.h)
file(READ "${ROOT}/codemp/rd-vulkan/vk_local.h" local_header)
string(FIND "${local_header}" "#define VK_CREATE_BUFFER(device" macros_start)
string(SUBSTRING "${local_header}" ${macros_start} -1 macros_tail)
string(FIND "${macros_tail}" "#endif // USE_VK_OBJECT_TRACKER" macros_end)
string(SUBSTRING "${macros_tail}" 0 ${macros_end} macros)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/allocation_macros.h" "${macros}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/codemp/rd-vulkan/vk_local.h")
add_executable(vulkan_resource_memory check.cpp)
target_include_directories(vulkan_resource_memory PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${PRODUCTION_ROOT}/codemp/rd-vulkan" "${ROOT}/codemp/rd-vulkan")
if(MSVC)
  target_compile_options(vulkan_resource_memory PRIVATE /EHsc)
endif()
