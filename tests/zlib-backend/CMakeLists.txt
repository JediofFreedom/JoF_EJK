cmake_minimum_required(VERSION 3.19)
project(zlib_backend_regressions C CXX)
set(CMAKE_CXX_STANDARD 11)
get_filename_component(ROOT "../.." ABSOLUTE)
option(UseZlibNG "Exercise the production zlib-ng backend" ON)
enable_testing()

# Deliberately hostile parent choices. They must remain local to the parent,
# while the runtime-dispatched dependency must always be portable and static.
set(BUILD_SHARED_LIBS ON)
set(BUILD_TESTING ON)
set(WITH_NATIVE_INSTRUCTIONS ON)
set(WITH_RUNTIME_CPU_DETECTION OFF)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_CUSTOM ON)
set(JKAInstallDir GameData)
set(JKAMPCoreComponent zlib-backend-license)
add_subdirectory("${ROOT}/lib/zlib" zlib)
if(NOT BUILD_SHARED_LIBS OR NOT BUILD_TESTING OR NOT WITH_NATIVE_INSTRUCTIONS OR WITH_RUNTIME_CPU_DETECTION)
  message(FATAL_ERROR "Dependency options escaped their scope")
endif()
if(NOT CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE OR NOT CMAKE_INTERPROCEDURAL_OPTIMIZATION_CUSTOM)
  message(FATAL_ERROR "Dependency changed parent LTO settings")
endif()
if(UseZlibNG)
  if(NOT EJK_ZLIB_BACKEND STREQUAL "zlib-ng" OR NOT TARGET zlib-ng)
    message(FATAL_ERROR "Default supported configuration did not select zlib-ng")
  endif()
  set(codec_target zlib-ng)
  foreach(config "" _DEBUG _RELEASE _RELWITHDEBINFO _MINSIZEREL _CUSTOM)
    get_target_property(ipo zlib-ng INTERPROCEDURAL_OPTIMIZATION${config})
    if(NOT ipo STREQUAL "OFF")
      message(FATAL_ERROR "zlib-ng IPO${config} is ${ipo}, expected explicit OFF")
    endif()
  endforeach()
else()
  if(NOT EJK_ZLIB_BACKEND STREQUAL "classic" OR TARGET zlib-ng)
    message(FATAL_ERROR "UseZlibNG=OFF did not retain classic zlib")
  endif()
  set(codec_target bundled_zlib)
endif()
get_target_property(codec_type ${codec_target} TYPE)
if(NOT codec_type STREQUAL "STATIC_LIBRARY")
  message(FATAL_ERROR "Bundled backend must remain static")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/parent.c" "int parent_policy(void) { return 0; }\n")
add_library(parent_policy STATIC EXCLUDE_FROM_ALL "${CMAKE_CURRENT_BINARY_DIR}/parent.c")
get_target_property(parent_ipo parent_policy INTERPROCEDURAL_OPTIMIZATION_RELEASE)
if(NOT parent_ipo)
  message(FATAL_ERROR "Parent Release target lost LTO")
endif()

# Standalone sanitizer runs need no LTO in their own fixture targets. The
# production parent-policy behavior has already been checked above.
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE OFF)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_CUSTOM OFF)
if(NOT UseZlibNG)
  set_property(TARGET bundled_zlib PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE OFF)
endif()
set(BUILD_SHARED_LIBS OFF)
add_subdirectory("${ROOT}/lib/minizip" minizip)
add_subdirectory("${ROOT}/lib/libpng" png)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
  target_compile_definitions(bundled_libpng PRIVATE PNG_ARM_NEON_OPT=0)
endif()
file(READ "${ROOT}/codemp/rd-common/tr_image_png.cpp" png)
string(FIND "${png}" "struct PNGReadContext" start)
string(FIND "${png}" "static void *PNG_Allocate" end)
math(EXPR length "${end}-${start}")
if(start LESS 0 OR length LESS 1)
  message(FATAL_ERROR "Production PNG decoder extraction failed")
endif()
string(SUBSTRING "${png}" ${start} ${length} decoder)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/png_decoder.h" "${decoder}\n")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/codemp/rd-common/tr_image_png.cpp")
add_executable(zlib_backend check.cpp)
target_include_directories(zlib_backend PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" ${ZLIB_INCLUDE_DIR}
  "${ROOT}/lib/minizip/include" "${ROOT}/lib/libpng/include" "${ROOT}/codemp/rd-common")
target_link_libraries(zlib_backend PRIVATE bundled_minizip bundled_libpng ${ZLIB_LIBRARIES})
target_compile_definitions(zlib_backend PRIVATE EXPECT_ZLIB_NG=$<BOOL:${UseZlibNG}>)
if(MSVC)
  target_compile_options(zlib_backend PRIVATE /EHsc)
  target_compile_definitions(zlib_backend PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/fixtures.py"
  "${CMAKE_CURRENT_BINARY_DIR}" COMMAND_ERROR_IS_FATAL ANY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/fixtures.py")
add_test(NAME zlib_backend_streams_zip_png COMMAND zlib_backend "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME zlib_backend_support_policy COMMAND "${CMAKE_COMMAND}" -DROOT=${ROOT}
  -P "${CMAKE_CURRENT_SOURCE_DIR}/policy.cmake")
set(fallback_arguments -G "${CMAKE_GENERATOR}"
  -DROOT=${ROOT} -DCMAKE_C_COMPILER=${CMAKE_C_COMPILER}
  -DCMAKE_MAKE_PROGRAM=${CMAKE_MAKE_PROGRAM} -DCMAKE_BUILD_TYPE=Release -DCMAKE_POLICY_VERSION_MINIMUM=3.5)
if(CMAKE_GENERATOR_PLATFORM)
  list(APPEND fallback_arguments -A "${CMAKE_GENERATOR_PLATFORM}")
endif()
if(CMAKE_GENERATOR_TOOLSET)
  list(APPEND fallback_arguments -T "${CMAKE_GENERATOR_TOOLSET}")
endif()
foreach(reason old_cmake unknown_compiler)
  add_test(NAME zlib_backend_fallback_${reason} COMMAND "${CMAKE_COMMAND}"
    -S "${CMAKE_CURRENT_SOURCE_DIR}/fallback" -B "${CMAKE_CURRENT_BINARY_DIR}/fallback-${reason}"
    ${fallback_arguments} -DFALLBACK=${reason})
endforeach()
if(UseZlibNG)
  add_test(NAME zlib_backend_license_install COMMAND "${CMAKE_COMMAND}"
    -DROOT=${ROOT} -DBUILD=${CMAKE_CURRENT_BINARY_DIR} -DCONFIG=$<CONFIG>
    -P "${CMAKE_CURRENT_SOURCE_DIR}/install.cmake")
endif()
