cmake_minimum_required(VERSION 3.19)
project(zlib_backend_fallback C)
set(UseZlibNG ON)
set(real_cmake_version "${CMAKE_VERSION}")
set(real_compiler_id "${CMAKE_C_COMPILER_ID}")
if(FALLBACK STREQUAL "old_cmake")
  # Exercise the actual chooser with the minimum parent CMake version. This
  # simulates its version input; it does not claim to run an old CMake binary.
  set(CMAKE_VERSION 3.10.3)
elseif(FALLBACK STREQUAL "unknown_compiler")
  set(CMAKE_C_COMPILER_ID UnverifiedVendor)
else()
  message(FATAL_ERROR "Unknown fallback fixture")
endif()
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
add_subdirectory("${ROOT}/lib/zlib" zlib)
if(NOT EJK_ZLIB_BACKEND STREQUAL "classic" OR NOT TARGET bundled_zlib OR TARGET zlib-ng)
  message(FATAL_ERROR "Unsupported configuration failed to select classic zlib")
endif()
get_target_property(ipo bundled_zlib INTERPROCEDURAL_OPTIMIZATION_RELEASE)
if(NOT ipo OR NOT EXISTS "${ZLIB_INCLUDE_DIR}/zlib.h")
  message(FATAL_ERROR "Classic fallback lost Release IPO (${ipo}) or public headers (${ZLIB_INCLUDE_DIR})")
endif()
set(CMAKE_VERSION "${real_cmake_version}")
set(CMAKE_C_COMPILER_ID "${real_compiler_id}")
