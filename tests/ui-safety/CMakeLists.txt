cmake_minimum_required(VERSION 3.19)
project(ui_safety CXX)
set(CMAKE_CXX_STANDARD 11)
get_filename_component(ROOT "../.." ABSOLUTE)
set(REFERENCE_REF "" CACHE STRING "Optional git reference to reproduce old failures")
enable_testing()

function(extract path signature output)
  if(REFERENCE_REF)
    execute_process(COMMAND git show "${REFERENCE_REF}:${path}" WORKING_DIRECTORY "${ROOT}"
      OUTPUT_VARIABLE source COMMAND_ERROR_IS_FATAL ANY)
  else()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/${path}")
    file(READ "${ROOT}/${path}" source)
  endif()
  string(FIND "${source}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing production function: ${signature}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} function)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/${output}.h" "${function}\n")
endfunction()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/previews.h" "")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/item.h" "")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/menu.h" "")
extract(codemp/ui/ui_main.c "static void UI_DrawMapPreview(" previews)
extract(codemp/ui/ui_main.c "static void UI_DrawNetMapPreview(" previews)
extract(codemp/ui/ui_shared.c "qboolean PC_Int_Parse(" item)
extract(codemp/ui/ui_shared.c "qboolean PC_String_Parse(" item)
extract(codemp/ui/ui_shared.c "qboolean ItemParse_cvarInt(" item)
extract(codemp/ui/ui_shared.c "void *UI_Alloc(" menu)
extract(codemp/ui/ui_shared.c "void Item_Init(" menu)
extract(codemp/ui/ui_shared.c "qboolean MenuParse_itemDef(" menu)
file(READ "${ROOT}/codemp/ui/ui_shared.h" types)
string(REGEX MATCH "typedef struct editFieldDef_s[^}]+} editFieldDef_t;" edit_type "${types}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/edit_type.h" "${edit_type}")
add_executable(ui_safety check.cpp)
target_include_directories(ui_safety PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
  # Keep successive preview calls in the same frame for the stale-handle
  # regression. This is compatible with AddressSanitizer.
  target_compile_options(ui_safety PRIVATE /Od /EHsc)
  target_compile_definitions(ui_safety PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
foreach(case preview-map preview-net cvar-int menu-allocation)
  add_test(NAME ui_${case} COMMAND ui_safety ${case})
endforeach()
