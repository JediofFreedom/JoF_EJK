cmake_minimum_required(VERSION 3.16)
project(sound_antispam_check CXX)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/../../codemp/client/snd_dma.cpp")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../codemp/client/snd_dma.cpp" source)
string(FIND "${source}" "qboolean S_CanPlaySound(" start)
string(SUBSTRING "${source}" ${start} -1 section)
string(FIND "${section}" "/*" end)
string(SUBSTRING "${section}" 0 ${end} section)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/limiter.h" "${section}")
add_executable(sound_antispam_check check.cpp)
target_include_directories(sound_antispam_check PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
enable_testing()
add_test(NAME sound_antispam COMMAND sound_antispam_check)
