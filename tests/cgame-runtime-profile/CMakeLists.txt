cmake_minimum_required(VERSION 3.16)
project(cgame_runtime_profile C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 11)
get_filename_component(ROOT "../.." ABSOLUTE)
find_package(Threads REQUIRED)
include("${CMAKE_CURRENT_SOURCE_DIR}/../TestPlatform.cmake")
configure_file("${ROOT}/codemp/versionfile/jofversion.h.in" jofversion.h @ONLY)
add_executable(cgame_runtime_profile check.cpp c_adapter.c anim_adapter.c "${ROOT}/codemp/cgame/cg_runtime_profile.cpp")
target_include_directories(cgame_runtime_profile PRIVATE "${ROOT}/codemp/cgame" "${CMAKE_CURRENT_BINARY_DIR}")
target_include_directories(cgame_runtime_profile PRIVATE "${ROOT}/codemp" "${ROOT}/shared")
target_compile_definitions(cgame_runtime_profile PRIVATE _CGAME)
ejk_test_platform(cgame_runtime_profile)
target_link_libraries(cgame_runtime_profile PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(cgame_runtime_profile PRIVATE /EHsc)
else()
  target_link_libraries(cgame_runtime_profile PRIVATE m)
endif()
foreach(path cg_view.c cg_main.c cg_players.c)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/codemp/cgame/${path}")
endforeach()
file(READ "${ROOT}/codemp/cgame/cg_players.c" players)
string(REGEX MATCH "void CG_Player\\( centity_t \\*cent \\) \\{[^}]+\\}" player_wrapper "${players}")
if(NOT player_wrapper MATCHES "CG_PlayerInternal\\(cent,&profile\\)")
  message(FATAL_ERROR "Actual player profiling wrapper was not extracted")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/player_wrapper.inc" "${player_wrapper}\n")
string(FIND "${players}" "void CG_TriggerAnimSounds(" trigger_start)
string(SUBSTRING "${players}" ${trigger_start} -1 trigger_tail)
string(FIND "${trigger_tail}" "\n}" trigger_end)
math(EXPR trigger_length "${trigger_end}+2")
string(SUBSTRING "${trigger_tail}" 0 ${trigger_length} trigger)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/anim_trigger.inc" "${trigger}\n")
string(FIND "${players}" "\t//trigger animation-based sounds, done before next lerp frame." animation_start)
string(SUBSTRING "${players}" ${animation_start} -1 animation_tail)
string(FIND "${animation_tail}" "\t// add the talk baloon or disconnect icon" animation_length)
if(trigger_start LESS 0 OR animation_start LESS 0 OR animation_length LESS 1)
  message(FATAL_ERROR "Actual animation profiling call site was not found")
endif()
string(SUBSTRING "${animation_tail}" 0 ${animation_length} animation)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/anim_callsite.inc" "${animation}\n")
file(READ "${ROOT}/codemp/cgame/cg_view.c" view)
string(REGEX MATCHALL "CG_EndRuntimeProfileFrame\\(profileFrame,0\\)" returns "${view}")
list(LENGTH returns return_count)
if(NOT return_count EQUAL 2)
  message(FATAL_ERROR "Both cgame loading/snapshot early returns must finish runtime profiling")
endif()
file(READ "${ROOT}/codemp/cgame/cg_main.c" main)
string(FIND "${main}" "CG_RuntimeProfileReady();\n}\n\n//makes sure returned string" ready)
if(ready LESS 0)
  message(FATAL_ERROR "Runtime profiler must become ready only at the end of CG_Init")
endif()
enable_testing()
add_test(NAME cgame_runtime_profile COMMAND cgame_runtime_profile)
