cmake_minimum_required(VERSION 3.16)
project(prepared_texture_cache CXX)
set(CMAKE_CXX_STANDARD 11)
find_package(Threads REQUIRED)
add_executable(prepared_texture_cache check.cpp ../../codemp/rd-vulkan/vk_prepared_cache.cpp)
target_include_directories(prepared_texture_cache PRIVATE ../../codemp/rd-vulkan "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(prepared_texture_cache PRIVATE Threads::Threads)
if(MSVC)
  target_compile_definitions(prepared_texture_cache PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(prepared_texture_cache PRIVATE /EHsc)
endif()
set(engine "${CMAKE_CURRENT_SOURCE_DIR}/../../codemp/qcommon/files.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${engine}")
file(READ "${engine}" source)
string(FIND "${source}" "qboolean FS_GetWritePath(" start)
if(start LESS 0)
  message(FATAL_ERROR "Missing native cache path helper")
endif()
string(SUBSTRING "${source}" ${start} -1 tail)
string(FIND "${tail}" "\n}" end)
math(EXPR length "${end}+2")
string(SUBSTRING "${tail}" 0 ${length} function)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/write_path.h" "${function}\n")
enable_testing()
add_test(NAME prepared_texture_cache COMMAND prepared_texture_cache "${CMAKE_CURRENT_BINARY_DIR}")
