cmake_minimum_required(VERSION 3.19)
project(vulkan_debug_capture CXX)
set(CMAKE_CXX_STANDARD 11)
get_filename_component(default_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(EJK_SOURCE_DIR "${default_root}" CACHE PATH "Engine source checkout")
set(DEBUG_SOURCE_DIR "${EJK_SOURCE_DIR}/codemp/rd-vulkan" CACHE PATH "Production sources; can select a pre-fix snapshot")
function(extract path signature output)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
  file(READ "${path}" source)
  string(FIND "${source}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing production definition: ${signature}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} body)
  set(${output} "${body}" PARENT_SCOPE)
endfunction()
file(READ "${EJK_SOURCE_DIR}/codemp/qcommon/qfiles.h" qfiles)
string(REGEX MATCH "#define[ \t]+SHADER_MAX_VERTEXES[^\n]*" vertex_limit "${qfiles}")
string(REGEX MATCH "#define[ \t]+SHADER_MAX_INDEXES[^\n]*" index_limit "${qfiles}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/capacity.h" "${vertex_limit}\n${index_limit}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/debug.h" "")
foreach(signature "void DrawNormals(" "static void transform_to_eye_space(" "static void R_DebugPolygon(")
  extract("${DEBUG_SOURCE_DIR}/vk_debug.cpp" "${signature}" body)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/debug.h" "${body}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/capture.h" "")
extract("${DEBUG_SOURCE_DIR}/tr_image.cpp" "void R_GammaCorrect(" body)
file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/capture.h" "${body}\n")
foreach(signature "byte *RB_ReadPixels(" "void R_TakeScreenshot(" "void R_TakeScreenshotPNG(" "void R_TakeScreenshotJPEG(" "static void R_LevelShot(" "const void *RB_TakeVideoFrameCmd(")
  extract("${DEBUG_SOURCE_DIR}/tr_init.cpp" "${signature}" body)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/capture.h" "${body}\n")
endforeach()
foreach(target vulkan_debug_capacity vulkan_capture_gamma)
  if(target STREQUAL "vulkan_debug_capacity")
    set(source debug.cpp)
  else()
    set(source capture.cpp)
  endif()
  add_executable(${target} ${source})
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${EJK_SOURCE_DIR}/codemp")
  if(MSVC)
    target_compile_options(${target} PRIVATE /EHsc)
  endif()
endforeach()
enable_testing()
add_test(NAME vulkan_debug_normals COMMAND vulkan_debug_capacity normals)
add_test(NAME vulkan_debug_polygons COMMAND vulkan_debug_capacity polygons)
foreach(format tga jpeg png levelshot video)
  add_test(NAME vulkan_capture_${format} COMMAND vulkan_capture_gamma ${format})
endforeach()
