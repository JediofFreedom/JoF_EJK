function(extract_function filename signature output)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${filename}")
  file(READ "${filename}" source)
  string(FIND "${source}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing ${signature}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  if(end LESS 0)
    message(FATAL_ERROR "Unterminated ${signature}")
  endif()
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} body)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}" "${body}\n")
endfunction()

foreach(kind storage descriptors indirect vertex)
  if(kind STREQUAL storage)
    set(signature "void vk_create_storage_buffer(")
  elseif(kind STREQUAL descriptors)
    set(signature "void vk_init_descriptors(")
  elseif(kind STREQUAL indirect)
    set(signature "void vk_create_indirect_buffer(")
  else()
    set(signature "void vk_create_vertex_buffer(")
  endif()
  extract_function("${VULKAN_SOURCE_DIRECTORY}/vk_shade_geometry.cpp" "${signature}" "${kind}.h")
endforeach()
