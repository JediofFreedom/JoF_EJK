cmake_minimum_required(VERSION 3.19)
project(navigation_loading CXX)
set(CMAKE_CXX_STANDARD 11)
enable_testing()
get_filename_component(ROOT "../.." ABSOLUTE)
set(REFERENCE_REF "" CACHE STRING "Optional git ref for baseline reproduction")
function(source path output)
  if(REFERENCE_REF)
    execute_process(COMMAND git show "${REFERENCE_REF}:${path}" WORKING_DIRECTORY "${ROOT}"
      OUTPUT_VARIABLE text COMMAND_ERROR_IS_FATAL ANY)
  else()
    file(READ "${ROOT}/${path}" text)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/${path}")
  endif()
  set(${output} "${text}" PARENT_SCOPE)
endfunction()
function(extract signature)
  string(FIND "${implementation}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing ${signature}")
  endif()
  string(SUBSTRING "${implementation}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} function)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/navigation.h" "${function}\n")
endfunction()
source(codemp/server/NPCNav/navigator.cpp implementation)
source(codemp/server/NPCNav/navigator.h declarations)
string(FIND "${declarations}" "class CNode\n" start)
string(SUBSTRING "${declarations}" ${start} -1 tail)
string(FIND "${tail}" "\n};" end)
math(EXPR length "${end}+3")
string(SUBSTRING "${tail}" 0 ${length} node)
# Expose fixture fields and count allocations without replacing production
# constructors/destructors or either serialization implementation.
string(REPLACE "\nprotected:\n" "\npublic:\n" node "${node}")
string(FIND "${node}" "public:" public_start)
math(EXPR hook_position "${public_start}+7")
string(SUBSTRING "${node}" 0 ${hook_position} prefix)
string(SUBSTRING "${node}" ${hook_position} -1 suffix)
set(node "${prefix}\n  static void *operator new(size_t size) { ++liveNodes; void *p = std::malloc(size); if (!p) throw std::bad_alloc(); return p; }\n  static void operator delete(void *p) { --liveNodes; std::free(p); }\n${suffix}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/node.h" "${node}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/navigation.h" "")
foreach(signature "CNode::CNode( void )" "CNode::~CNode( void )" "CNode *CNode::Create( void )"
    "void CNode::InitRanks(" "int\tCNode::Save(" "int CNode::Load("
    "void CNavigator::Free(" "bool CNavigator::Load(" "bool CNavigator::Save(")
  extract("${signature}")
endforeach()
string(FIND "${implementation}" "int\tCNavigator::GetInt(" legacy_start)
if(NOT legacy_start LESS 0)
  extract("int\tCNavigator::GetInt(")
  extract("long CNavigator::GetLong(")
endif()
add_executable(navigation_loading check.cpp)
target_include_directories(navigation_loading PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${ROOT}/codemp/server/NPCNav")
if(MSVC)
  target_compile_options(navigation_loading PRIVATE /EHsc)
  target_compile_definitions(navigation_loading PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_test(NAME nav_roundtrip COMMAND navigation_loading valid)
add_test(NAME nav_partial_cleanup COMMAND navigation_loading ownership)
add_test(NAME nav_truncations COMMAND navigation_loading truncations)
add_test(NAME nav_invalid_tables COMMAND navigation_loading invalid)
