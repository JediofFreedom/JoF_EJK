cmake_minimum_required(VERSION 3.16)
project(cgame_lead_projection CXX)
set(CMAKE_CXX_STANDARD 11)
if(NOT ROOT)
  get_filename_component(ROOT "../.." ABSOLUTE)
endif()
set(DRAW_SOURCE "${ROOT}/codemp/cgame/cg_draw.c" CACHE FILEPATH "Actual cgame source; may select a pre-fix snapshot")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${DRAW_SOURCE}")
file(READ "${DRAW_SOURCE}" source)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/projection.h" "")
set(weapon_source "${ROOT}/codemp/cgame/cg_weapons.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${weapon_source}")
file(READ "${weapon_source}" weapons)
foreach(signature "int CG_GetBulletSpeed(" "qboolean CG_GetBulletGravity(")
  string(FIND "${weapons}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing actual weapon helper ${signature}")
  endif()
  string(SUBSTRING "${weapons}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} function)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/projection.h" "${function}\n")
endforeach()
foreach(signature "qboolean CG_WorldCoordToScreenCoord(vec3_t" "void CG_DottedLine( float" "void CG_BracketEntity( centity_t" "static void CG_LeadIndicator(void)")
  string(FIND "${source}" "${signature}" start REVERSE)
  if(start LESS 0)
    message(FATAL_ERROR "Missing actual function ${signature}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} function)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/projection.h" "${function}\n")
endforeach()
add_executable(cgame_lead_projection "${ROOT}/tests/cgame-lead-projection/check.cpp")
target_include_directories(cgame_lead_projection PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
  target_compile_options(cgame_lead_projection PRIVATE /EHsc)
endif()
enable_testing()
foreach(case projector player-valid vehicle-valid player-off-viewport vehicle-off-viewport
             player-behind vehicle-behind player-near vehicle-near player-empty vehicle-empty
             player-origin-behind vehicle-origin-behind player-mixed)
  add_test(NAME lead_${case} COMMAND cgame_lead_projection ${case})
endforeach()
