cmake_minimum_required(VERSION 3.16)
project(sound_start_profile CXX)
set(CMAKE_CXX_STANDARD 11)
include("${CMAKE_CURRENT_SOURCE_DIR}/../TestPlatform.cmake")
get_filename_component(ROOT "../.." ABSOLUTE)
configure_file("${ROOT}/codemp/versionfile/jofversion.h.in" jofversion.h @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/codemp/client/snd_dma.cpp" "${ROOT}/codemp/client/snd_local.h")
file(READ "${ROOT}/codemp/client/snd_dma.cpp" source)
string(FIND "${source}" "namespace SoundStartProfile {" first)
string(FIND "${source}" "// End sound submission profiling." last)
math(EXPR length "${last}-${first}")
string(SUBSTRING "${source}" ${first} ${length} profile)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/profile.inc" "${profile}\n")
string(FIND "${source}" "void S_StartSound(const vec3_t origin," start)
string(SUBSTRING "${source}" ${start} -1 tail)
string(FIND "${tail}" "\n}" end)
math(EXPR length "${end}+2")
string(SUBSTRING "${tail}" 0 ${length} function)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/start.inc" "${function}\n")
file(READ "${ROOT}/codemp/client/snd_local.h" types)
string(FIND "${types}" "typedef enum\n{\n\tct_16" first)
string(FIND "${types}" "typedef struct soundBuffer_s" last)
math(EXPR length "${last}-${first}")
string(SUBSTRING "${types}" ${first} ${length} sound_types)
string(FIND "${types}" "#define START_SAMPLE_IMMEDIATE" first)
string(FIND "${types}" "#define\tWAV_FORMAT_PCM" last)
math(EXPR length "${last}-${first}")
string(SUBSTRING "${types}" ${first} ${length} channel_types)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/types.inc" "${sound_types}\n${channel_types}\n")
foreach(required "Cvar_Get(\"s_runtimeProfile\", \"0\"" "Cmd_AddCommand(\"s_runtimeProfileReset\"" "Cmd_AddCommand(\"s_runtimeProfileReport\"" "Cmd_RemoveCommand(\"s_runtimeProfileReset\"" "Cmd_RemoveCommand(\"s_runtimeProfileReport\"")
  string(FIND "${source}" "${required}" found)
  if(found LESS 0)
    message(FATAL_ERROR "Missing sound profiler lifecycle action: ${required}")
  endif()
endforeach()
add_executable(sound_start_profile check.cpp)
target_include_directories(sound_start_profile PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${ROOT}/codemp" "${ROOT}/shared")
ejk_test_platform(sound_start_profile)
if(MSVC)
  target_compile_options(sound_start_profile PRIVATE /EHsc)
endif()
enable_testing()
add_test(NAME sound_start_profile COMMAND sound_start_profile)
