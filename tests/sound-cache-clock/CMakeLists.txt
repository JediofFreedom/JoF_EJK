cmake_minimum_required(VERSION 3.19)
project(sound_cache_clock CXX)
set(CMAKE_CXX_STANDARD 11)
include("${CMAKE_CURRENT_SOURCE_DIR}/../TestPlatform.cmake")
get_filename_component(ROOT "../.." ABSOLUTE)
configure_file("${ROOT}/codemp/versionfile/jofversion.h.in" jofversion.h @ONLY)
set(REFERENCE_REF "" CACHE STRING "Optional git ref supplying the subject touch for avoided-work verification")
function(extract path signature output)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/${path}")
  file(READ "${ROOT}/${path}" source)
  if(REFERENCE_REF AND output STREQUAL "touch.inc")
    execute_process(COMMAND git show "${REFERENCE_REF}:${path}" WORKING_DIRECTORY "${ROOT}"
      OUTPUT_VARIABLE source COMMAND_ERROR_IS_FATAL ANY)
  endif()
  string(FIND "${source}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing production function: ${signature}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} body)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/${output}" "${body}\n")
endfunction()
foreach(output touch.inc cache.inc clocks.inc)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}" "")
endforeach()
extract(codemp/client/snd_dma.cpp "void SND_TouchSFX(sfx_t *sfx)" touch.inc)
foreach(signature "static int SND_FreeSFXMem(sfx_t *sfx)\n{" "int SND_FreeOldestSound(sfx_t *pButNotThisOne" "qboolean SND_RegisterAudio_LevelLoadEnd(")
  extract(codemp/client/snd_dma.cpp "${signature}" cache.inc)
endforeach()
foreach(signature "sysEvent_t\tCom_GetRealEvent(" "void Com_InitPushEvent(" "void Com_PushEvent(" "sysEvent_t\tCom_GetEvent(" "int Com_Milliseconds (")
  extract(codemp/qcommon/common.cpp "${signature}" clocks.inc)
endforeach()
foreach(signature "static const char *Sys_EventName(" "sysEvent_t Sys_GetEvent(" "void Sys_QueEvent(")
  extract(shared/sys/sys_event.cpp "${signature}" clocks.inc)
endforeach()
file(READ "${ROOT}/codemp/client/snd_local.h" types)
string(FIND "${types}" "typedef enum\n{\n\tct_16" first)
string(FIND "${types}" "typedef struct soundBuffer_s" last)
math(EXPR length "${last}-${first}")
string(SUBSTRING "${types}" ${first} ${length} sound_types)
string(FIND "${types}" "#define START_SAMPLE_IMMEDIATE" first)
string(FIND "${types}" "#define\tWAV_FORMAT_PCM" last)
math(EXPR length "${last}-${first}")
string(SUBSTRING "${types}" ${first} ${length} channel_types)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/types.inc" "${sound_types}\n${channel_types}\n#define MAX_CHANNELS 32\n")
add_executable(sound_cache_clock check.cpp)
target_include_directories(sound_cache_clock PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${ROOT}/codemp" "${ROOT}/shared")
