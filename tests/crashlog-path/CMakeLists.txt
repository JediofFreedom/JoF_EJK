cmake_minimum_required(VERSION 3.19)
project(crashlog_path CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(default_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(EJK_SOURCE_DIR "${default_root}" CACHE PATH "Engine checkout")
set(SYS_MAIN_SOURCE "${EJK_SOURCE_DIR}/shared/sys/sys_main.cpp" CACHE FILEPATH "Production source; may select pre-fix snapshot")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SYS_MAIN_SOURCE}")
file(READ "${SYS_MAIN_SOURCE}" source)
string(FIND "${source}" "static void Sys_ErrorDialog(" start)
if(start LESS 0)
  message(FATAL_ERROR "Missing actual production Sys_ErrorDialog")
endif()
string(SUBSTRING "${source}" ${start} -1 tail)
string(FIND "${tail}" "\n}" end)
if(end LESS 0)
  message(FATAL_ERROR "Missing production function terminator")
endif()
math(EXPR length "${end}+2")
string(SUBSTRING "${tail}" 0 ${length} body)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/actual_crashlog.h" "${body}\n")
add_executable(crashlog_path check.cpp)
target_include_directories(crashlog_path PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
  target_compile_options(crashlog_path PRIVATE /EHsc)
endif()
enable_testing()
foreach(part portable explicit normal_default default_null empty all_null)
  add_test(NAME crashlog_path_${part} COMMAND crashlog_path ${part})
endforeach()
