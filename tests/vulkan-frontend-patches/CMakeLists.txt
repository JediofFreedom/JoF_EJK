cmake_minimum_required(VERSION 3.16)
project(vulkan_frontend_patches CXX)
set(CMAKE_CXX_STANDARD 11)
set(VULKAN_SOURCE_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../codemp/rd-vulkan" CACHE PATH "Production renderer source")
set(path "${VULKAN_SOURCE_DIRECTORY}/tr_bsp.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
file(READ "${path}" source)
set(functions "")
foreach(signature "static int R_MergedWidthPoints(" "static int R_MergedHeightPoints(" "static int R_StitchPatches(" "static void R_MovePatchSurfacesToHunk(")
  string(FIND "${source}" "${signature}" start)
  if(start LESS 0)
    message(FATAL_ERROR "Missing ${signature}")
  endif()
  string(SUBSTRING "${source}" ${start} -1 tail)
  string(FIND "${tail}" "\n}" end)
  math(EXPR length "${end}+2")
  string(SUBSTRING "${tail}" 0 ${length} body)
  string(APPEND functions "${body}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/patch-functions.h" "${functions}")
add_executable(patch_stitch check.cpp)
target_include_directories(patch_stitch PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
enable_testing()
add_test(NAME patch_stitch COMMAND patch_stitch stitch)
add_test(NAME patch_move COMMAND patch_stitch move)
